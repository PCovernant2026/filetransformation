




/// <summary>
/// ============================================================
///  PLUGIN 1:
///  OpportunityClientPartnerFromUserFlagPlugin
///
///  Business rule:
///  - On Opportunity Create
///  - Query the creating user
///  - If systemuser.new_clientpartner = Yes/true
///  - Populate opportunity.new_clientpartnerid
///
///  Register:
///  - Table: opportunity
///  - Message: Create
///  - Stage: PreOperation
///  - Mode: Synchronous
///  - Filtering Attributes: none
///  - Images: none
/// ============================================================
/// </summary>
public class OpportunityClientPartnerFromUserFlagPlugin : PluginBase
{
    protected override void ExecuteInternal(
        IPluginExecutionContext context,
        IOrganizationService service,
        ITracingService trace)
    {
        trace?.Trace("TRACE-PARTNER: Starting OpportunityClientPartnerFromUserFlagPlugin.");

        if (context.Depth > 1)
        {
            trace?.Trace("TRACE-PARTNER-SKIP: Depth {0} > 1. Exiting.", context.Depth);
            return;
        }

        if (!IsPreOperation(context) ||
            !IsMessage(context, "Create") ||
            !IsPrimaryEntity(context, Schema.Opportunity))
        {
            trace?.Trace(
                "TRACE-PARTNER-SKIP: Expected Create opportunity PreOperation. Actual Message={0}; Entity={1}; Stage={2}.",
                context.MessageName,
                context.PrimaryEntityName,
                context.Stage);
            return;
        }

        var target = GetTargetEntity(context);
        TraceTarget(trace, target);

        if (target == null)
        {
            trace?.Trace("TRACE-PARTNER-SKIP: Target is null.");
            return;
        }

        if (target.Attributes.Contains(Schema.OpportunityClientPartner) &&
            target[Schema.OpportunityClientPartner] != null)
        {
            trace?.Trace(
                "TRACE-PARTNER-NOOP: {0} already supplied as {1}. No overwrite.",
                Schema.OpportunityClientPartner,
                FormatEntityReference(target.GetAttributeValue<EntityReference>(Schema.OpportunityClientPartner)));
            return;
        }

        var creatingUserId = context.InitiatingUserId != Guid.Empty
            ? context.InitiatingUserId
            : context.UserId;

        if (creatingUserId == Guid.Empty)
        {
            trace?.Trace("TRACE-PARTNER-SKIP: Creating user could not be resolved.");
            return;
        }

        trace?.Trace(
            "TRACE-PARTNER: Retrieving systemuser {0} with column {1}.",
            creatingUserId,
            Schema.SystemUserIsClientPartner);

        var creatingUser = service.Retrieve(
            Schema.SystemUser,
            creatingUserId,
            new ColumnSet(Schema.SystemUserIsClientPartner));

        if (creatingUser == null)
        {
            trace?.Trace("TRACE-PARTNER-SKIP: Creating user record could not be retrieved.");
            return;
        }

        bool isClientPartner =
            creatingUser.Contains(Schema.SystemUserIsClientPartner) &&
            creatingUser.GetAttributeValue<bool>(Schema.SystemUserIsClientPartner);

        trace?.Trace(
            "TRACE-PARTNER: Creating user {0}; {1}={2}.",
            creatingUserId,
            Schema.SystemUserIsClientPartner,
            isClientPartner);

        if (!isClientPartner)
        {
            trace?.Trace(
                "TRACE-PARTNER-NOOP: User is not marked as Client Partner. {0} not populated.",
                Schema.OpportunityClientPartner);
            return;
        }

        var userRef = new EntityReference(Schema.SystemUser, creatingUserId);
        target[Schema.OpportunityClientPartner] = userRef;

        trace?.Trace(
            "TRACE-PARTNER-ACTION: User is Client Partner. Defaulted {0} to {1}.",
            Schema.OpportunityClientPartner,
            FormatEntityReference(userRef));
    }
}

public class OpportunityWonStagePlugin : PluginBase
{
    protected override void ExecuteInternal(
        IPluginExecutionContext context,
        IOrganizationService service,
        ITracingService trace)
    {
        trace?.Trace("TRACE-WONSTAGE: Starting OpportunityWonStagePlugin.");

        if (context.Depth > 1)
        {
            trace?.Trace("TRACE-WONSTAGE-SKIP: Depth {0} > 1. Exiting.", context.Depth);
            return;
        }

        if (IsMessage(context, "Update"))
        {
            HandleUpdatePath(context, trace);
            return;
        }

        // IMPORTANT:
        // Your org fires Message=Win, not WinOpportunity.
        // We support both to be safe.
        if (IsMessage(context, "Win") || IsMessage(context, "WinOpportunity"))
        {
            HandleWinPath(context, service, trace);
            return;
        }

        if (IsMessage(context, "SetState") ||
            IsMessage(context, "SetStateDynamicEntity"))
        {
            HandleSetStatePath(context, service, trace);
            return;
        }

        trace?.Trace(
            "TRACE-WONSTAGE-SKIP: Message {0} is not handled.",
            context.MessageName);
    }

    private static void HandleUpdatePath(
        IPluginExecutionContext context,
        ITracingService trace)
    {
        trace?.Trace("TRACE-WONSTAGE: Handling Update path.");

        if (!IsPreOperation(context) ||
            !IsPrimaryEntity(context, Schema.Opportunity))
        {
            trace?.Trace(
                "TRACE-WONSTAGE-SKIP: Update path expected opportunity PreOperation. Actual Entity={0}; Stage={1}.",
                context.PrimaryEntityName,
                context.Stage);
            return;
        }

        var target = GetTargetEntity(context);
        TraceTarget(trace, target);

        if (target == null)
        {
            trace?.Trace("TRACE-WONSTAGE-SKIP: Target is null.");
            return;
        }

        var preImage = GetPreImage(context);
        TracePreImage(trace, preImage);

        var targetState = target.GetAttributeValue<OptionSetValue>(
            Schema.OpportunityStateCode);

        var preImageState = preImage?.GetAttributeValue<OptionSetValue>(
            Schema.OpportunityStateCode);

        var resolvedState = GetAttributeValueFromTargetOrPreImage<OptionSetValue>(
            target,
            preImage,
            Schema.OpportunityStateCode);

        trace?.Trace(
            "TRACE-WONSTAGE: TargetState={0}; PreImageState={1}; ResolvedState={2}.",
            FormatOptionSet(targetState),
            FormatOptionSet(preImageState),
            FormatOptionSet(resolvedState));

        if (resolvedState == null ||
            resolvedState.Value != Schema.OpportunityStateWon)
        {
            trace?.Trace(
                "TRACE-WONSTAGE-NOOP: Opportunity is not Won. Expected={0}; Actual={1}.",
                Schema.OpportunityStateWon,
                FormatOptionSet(resolvedState));
            return;
        }

        SetOpportunityStageTo100Percent(target);

        trace?.Trace(
            "TRACE-WONSTAGE-ACTION: Update path set {0}={1}.",
            Schema.OpportunityStage,
            Schema.OpportunityStage100Percent);
    }

    private static void HandleWinPath(
        IPluginExecutionContext context,
        IOrganizationService service,
        ITracingService trace)
    {
        trace?.Trace(
            "TRACE-WONSTAGE: Handling Win path. Message={0}; Stage={1}.",
            context.MessageName,
            context.Stage);

        Entity opportunityClose = null;

        if (context.InputParameters != null &&
            context.InputParameters.Contains(Schema.OpportunityCloseInputParameter) &&
            context.InputParameters[Schema.OpportunityCloseInputParameter] is Entity closeEntity)
        {
            opportunityClose = closeEntity;
        }

        if (opportunityClose == null)
        {
            trace?.Trace("TRACE-WONSTAGE-SKIP: OpportunityClose input was not found.");
            return;
        }

        trace?.Trace(
            "TRACE-WONSTAGE: OpportunityClose found. LogicalName={0}; Attributes=[{1}].",
            opportunityClose.LogicalName,
            FormatKeys(opportunityClose.Attributes));

        var opportunityRef = opportunityClose.GetAttributeValue<EntityReference>(
            Schema.OpportunityCloseOpportunity);

        trace?.Trace(
            "TRACE-WONSTAGE: Opportunity reference={0}.",
            FormatEntityReference(opportunityRef));

        if (opportunityRef == null ||
            opportunityRef.Id == Guid.Empty)
        {
            trace?.Trace("TRACE-WONSTAGE-SKIP: Invalid opportunity reference.");
            return;
        }

        var update = new Entity(Schema.Opportunity, opportunityRef.Id);
        SetOpportunityStageTo100Percent(update);

        service.Update(update);

        trace?.Trace(
            "TRACE-WONSTAGE-ACTION: Win path updated opportunity {0}. Set {1}={2}.",
            opportunityRef.Id,
            Schema.OpportunityStage,
            Schema.OpportunityStage100Percent);
    }

    private static void HandleSetStatePath(
        IPluginExecutionContext context,
        IOrganizationService service,
        ITracingService trace)
    {
        trace?.Trace("TRACE-WONSTAGE: Handling SetState / SetStateDynamicEntity path.");

        if (!IsPostOperation(context))
        {
            trace?.Trace(
                "TRACE-WONSTAGE-SKIP: SetState must be PostOperation stage 40. Actual Stage={0}.",
                context.Stage);
            return;
        }

        EntityReference entityMoniker = null;
        OptionSetValue state = null;

        if (context.InputParameters != null &&
            context.InputParameters.Contains(Schema.EntityMonikerInputParameter) &&
            context.InputParameters[Schema.EntityMonikerInputParameter] is EntityReference moniker)
        {
            entityMoniker = moniker;
        }

        if (context.InputParameters != null &&
            context.InputParameters.Contains(Schema.StateInputParameter) &&
            context.InputParameters[Schema.StateInputParameter] is OptionSetValue stateValue)
        {
            state = stateValue;
        }

        trace?.Trace(
            "TRACE-WONSTAGE: EntityMoniker={0}; State={1}.",
            FormatEntityReference(entityMoniker),
            FormatOptionSet(state));

        if (entityMoniker == null ||
            entityMoniker.Id == Guid.Empty ||
            !string.Equals(entityMoniker.LogicalName, Schema.Opportunity, StringComparison.OrdinalIgnoreCase))
        {
            trace?.Trace("TRACE-WONSTAGE-SKIP: EntityMoniker is not a valid opportunity.");
            return;
        }

        if (state == null ||
            state.Value != Schema.OpportunityStateWon)
        {
            trace?.Trace(
                "TRACE-WONSTAGE-NOOP: State is not Won. Expected={0}; Actual={1}.",
                Schema.OpportunityStateWon,
                FormatOptionSet(state));
            return;
        }

        var update = new Entity(Schema.Opportunity, entityMoniker.Id);
        SetOpportunityStageTo100Percent(update);

        service.Update(update);

        trace?.Trace(
            "TRACE-WONSTAGE-ACTION: SetState updated opportunity {0}. Set {1}={2}.",
            entityMoniker.Id,
            Schema.OpportunityStage,
            Schema.OpportunityStage100Percent);
    }
}
