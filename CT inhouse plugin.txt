    #region Shared Plugin Base

    /// <summary>
    /// Shared base plugin class.
    /// Handles common service setup, tracing, target extraction, image extraction, and helper formatting.
    /// </summary>
    public abstract class PluginBase : IPlugin
    {
        public void Execute(IServiceProvider serviceProvider)
        {
            if (serviceProvider == null)
            {
                throw new InvalidPluginExecutionException("Service provider is null.");
            }

            var context = (IPluginExecutionContext)serviceProvider.GetService(typeof(IPluginExecutionContext));
            var trace = (ITracingService)serviceProvider.GetService(typeof(ITracingService));
            var serviceFactory = (IOrganizationServiceFactory)serviceProvider.GetService(typeof(IOrganizationServiceFactory));

            if (context == null)
            {
                throw new InvalidPluginExecutionException("Plugin execution context is null.");
            }

            if (serviceFactory == null)
            {
                throw new InvalidPluginExecutionException("Organization service factory is null.");
            }

            // System context is used for reads/defaulting so user read privileges do not block automation.
            // Field changes are made through Target in PreOperation or explicit Update only where required.
            var service = serviceFactory.CreateOrganizationService(null);

            try
            {
                TraceExecutionStart(context, trace, GetType().FullName);

                ExecuteInternal(context, service, trace);

                trace?.Trace(
                    "TRACE-END: {0} completed successfully. CorrelationId={1}",
                    GetType().FullName,
                    context.CorrelationId);
            }
            catch (InvalidPluginExecutionException ex)
            {
                trace?.Trace(
                    "TRACE-ERROR: InvalidPluginExecutionException in {0}. Message={1}. CorrelationId={2}",
                    GetType().FullName,
                    ex.Message,
                    context.CorrelationId);

                throw;
            }
            catch (Exception ex)
            {
                trace?.Trace(
                    "TRACE-ERROR: Unhandled exception in {0}. CorrelationId={1}. Exception={2}",
                    GetType().FullName,
                    context.CorrelationId,
                    ex);

                throw new InvalidPluginExecutionException($"An error occurred in {GetType().Name}: {ex.Message}", ex);
            }
        }

        protected abstract void ExecuteInternal(
            IPluginExecutionContext context,
            IOrganizationService service,
            ITracingService trace);

        #region Context Helpers

        protected static Entity GetTargetEntity(IPluginExecutionContext context)
        {
            if (context.InputParameters != null &&
                context.InputParameters.Contains("Target") &&
                context.InputParameters["Target"] is Entity target)
            {
                return target;
            }

            return null;
        }

        protected static Entity GetPreImage(IPluginExecutionContext context, string imageName = "PreImage")
        {
            if (context.PreEntityImages != null &&
                context.PreEntityImages.Contains(imageName))
            {
                return context.PreEntityImages[imageName];
            }

            return null;
        }

        protected static T GetAttributeValueFromTargetOrPreImage<T>(
            Entity target,
            Entity preImage,
            string attributeName)
        {
            if (target != null && target.Attributes.Contains(attributeName))
            {
                return target.GetAttributeValue<T>(attributeName);
            }

            if (preImage != null && preImage.Attributes.Contains(attributeName))
            {
                return preImage.GetAttributeValue<T>(attributeName);
            }

            return default(T);
        }

        protected static bool IsMessage(IPluginExecutionContext context, string messageName)
        {
            return string.Equals(context.MessageName, messageName, StringComparison.OrdinalIgnoreCase);
        }

        protected static bool IsPrimaryEntity(IPluginExecutionContext context, string entityName)
        {
            return string.Equals(context.PrimaryEntityName, entityName, StringComparison.OrdinalIgnoreCase);
        }

        protected static bool IsPreOperation(IPluginExecutionContext context)
        {
            return context.Stage == Schema.PreOperationStage;
        }

        protected static bool IsPostOperation(IPluginExecutionContext context)
        {
            return context.Stage == Schema.PostOperationStage;
        }

        protected static bool IsDepthGreaterThanOne(IPluginExecutionContext context)
        {
            return context.Depth > 1;
        }

        #endregion

        #region Value Helpers

        protected static bool? GetNullableBoolean(Entity entity, string attributeName)
        {
            if (entity == null ||
                !entity.Attributes.Contains(attributeName) ||
                entity[attributeName] == null)
            {
                return null;
            }

            return entity.GetAttributeValue<bool>(attributeName);
        }

        protected static void SetWonProbabilityFields(Entity opportunity)
        {
            if (opportunity == null)
            {
                return;
            }

            opportunity[Schema.OpportunityProbability] = 100;
            opportunity[Schema.OpportunityCustomCloseProbability] =
                new OptionSetValue(Schema.CustomCloseProbabilityClosed);
        }

        protected static bool HasNonNullAttribute(Entity entity, string attributeName)
        {
            return entity != null &&
                   entity.Attributes.Contains(attributeName) &&
                   entity[attributeName] != null;
        }

        #endregion

        #region Trace Helpers

        protected static void TraceExecutionStart(
            IPluginExecutionContext context,
            ITracingService trace,
            string pluginName)
        {
            trace?.Trace("TRACE-START: {0}", pluginName);

            trace?.Trace(
                "TRACE-CONTEXT: Message={0}; Entity={1}; EntityId={2}; Stage={3}; Mode={4}; Depth={5}; UserId={6}; InitiatingUserId={7}; CorrelationId={8}; OperationId={9}",
                context.MessageName,
                context.PrimaryEntityName,
                context.PrimaryEntityId,
                context.Stage,
                context.Mode,
                context.Depth,
                context.UserId,
                context.InitiatingUserId,
                context.CorrelationId,
                context.OperationId);

            trace?.Trace(
                "TRACE-PARAMETERS: InputParameters=[{0}]; OutputParameters=[{1}]; SharedVariables=[{2}]",
                FormatKeys(context.InputParameters),
                FormatKeys(context.OutputParameters),
                FormatKeys(context.SharedVariables));
        }

        protected static void TraceTarget(ITracingService trace, Entity target)
        {
            if (target == null)
            {
                trace?.Trace("TRACE-TARGET: Target is null.");
                return;
            }

            trace?.Trace(
                "TRACE-TARGET: LogicalName={0}; Id={1}; AttributeCount={2}; Attributes=[{3}]",
                target.LogicalName,
                target.Id,
                target.Attributes.Count,
                FormatKeys(target.Attributes));
        }

        protected static void TracePreImage(
            ITracingService trace,
            Entity preImage,
            string imageName = "PreImage")
        {
            if (preImage == null)
            {
                trace?.Trace("TRACE-PREIMAGE: {0} was not available.", imageName);
                return;
            }

            trace?.Trace(
                "TRACE-PREIMAGE: {0}; LogicalName={1}; Id={2}; AttributeCount={3}; Attributes=[{4}]",
                imageName,
                preImage.LogicalName,
                preImage.Id,
                preImage.Attributes.Count,
                FormatKeys(preImage.Attributes));
        }

        /// <summary>
        /// Single collection formatter used for ParameterCollection and AttributeCollection.
        /// This avoids ambiguity between System.ComponentModel.AttributeCollection and Microsoft.Xrm.Sdk.AttributeCollection.
        /// </summary>
        protected static string FormatKeys(Microsoft.Xrm.Sdk.DataCollection<string, object> collection)
        {
            if (collection == null || collection.Count == 0)
            {
                return "none";
            }

            return string.Join(", ", collection.Keys);
        }

        protected static string FormatEntityReference(EntityReference entityReference)
        {
            if (entityReference == null)
            {
                return "null";
            }

            return $"{entityReference.LogicalName}:{entityReference.Id}";
        }

        protected static string FormatOptionSet(OptionSetValue optionSetValue)
        {
            return optionSetValue == null ? "null" : optionSetValue.Value.ToString();
        }

        #endregion
    }

    #region Plugin 1 - Opportunity Ownership Defaults

    /// <summary>
    /// Defaults Opportunity ownership fields when an Opportunity is created.
    ///
    /// Business rules:
    /// 1. If creating user has systemuser.new_clientmanager = true,
    ///    default opportunity.ownerid to the creating user when ownerid is blank.
    /// 2. If creating user has systemuser.new_clientpartner = true,
    ///    default opportunity.new_clientpartnerid to the creating user when blank.
    /// 3. If user has both flags, both fields are defaulted if blank.
    ///
    /// Register:
    /// - Table: opportunity
    /// - Message: Create
    /// - Stage: PreOperation
    /// - Mode: Synchronous
    /// - Filtering Attributes: none
    /// - Images: none
    /// </summary>
    public class OpportunityOwnershipDefaultsPlugin : PluginBase
    {
        protected override void ExecuteInternal(
            IPluginExecutionContext context,
            IOrganizationService service,
            ITracingService trace)
        {
            trace?.Trace("TRACE-PLUGIN1: Starting OpportunityOwnershipDefaultsPlugin.");

            if (IsDepthGreaterThanOne(context))
            {
                trace?.Trace("TRACE-PLUGIN1-SKIP: Depth {0} > 1. Exiting to avoid recursion.", context.Depth);
                return;
            }

            if (!IsValidCreateOpportunityPreOperation(context))
            {
                trace?.Trace(
                    "TRACE-PLUGIN1-SKIP: Expected Create opportunity PreOperation. Actual Message={0}, Entity={1}, Stage={2}.",
                    context.MessageName,
                    context.PrimaryEntityName,
                    context.Stage);
                return;
            }

            var target = GetTargetEntity(context);
            TraceTarget(trace, target);

            if (target == null)
            {
                trace?.Trace("TRACE-PLUGIN1-SKIP: Target was not an Entity.");
                return;
            }

            var creatingUserId = ResolveCreatingUserId(context, trace);
            if (creatingUserId == Guid.Empty)
            {
                trace?.Trace("TRACE-PLUGIN1-SKIP: Creating user could not be resolved.");
                return;
            }

            var user = RetrieveUserRoleFlags(service, creatingUserId, trace);
            if (user == null)
            {
                trace?.Trace("TRACE-PLUGIN1-SKIP: System user {0} could not be retrieved.", creatingUserId);
                return;
            }

            ApplyOwnershipDefaults(target, user, creatingUserId, trace);

            trace?.Trace("TRACE-PLUGIN1: Completed OpportunityOwnershipDefaultsPlugin.");
        }

        private static bool IsValidCreateOpportunityPreOperation(IPluginExecutionContext context)
        {
            return IsPreOperation(context) &&
                   IsMessage(context, "Create") &&
                   IsPrimaryEntity(context, Schema.Opportunity);
        }

        private static Guid ResolveCreatingUserId(IPluginExecutionContext context, ITracingService trace)
        {
            var creatingUserId = context.InitiatingUserId != Guid.Empty
                ? context.InitiatingUserId
                : context.UserId;

            trace?.Trace(
                "TRACE-PLUGIN1: CreatingUserId resolved to {0}. InitiatingUserId={1}; UserId={2}.",
                creatingUserId,
                context.InitiatingUserId,
                context.UserId);

            return creatingUserId;
        }

        private static Entity RetrieveUserRoleFlags(
            IOrganizationService service,
            Guid userId,
            ITracingService trace)
        {
            trace?.Trace(
                "TRACE-PLUGIN1: Retrieving systemuser {0} columns [{1}, {2}].",
                userId,
                Schema.UserIsClientManager,
                Schema.UserIsClientPartner);

            return service.Retrieve(
                Schema.SystemUser,
                userId,
                new ColumnSet(
                    Schema.UserIsClientManager,
                    Schema.UserIsClientPartner));
        }

        private static void ApplyOwnershipDefaults(
            Entity target,
            Entity user,
            Guid creatingUserId,
            ITracingService trace)
        {
            var isClientManagerNullable = GetNullableBoolean(user, Schema.UserIsClientManager);
            var isClientPartnerNullable = GetNullableBoolean(user, Schema.UserIsClientPartner);

            var isClientManager = isClientManagerNullable == true;
            var isClientPartner = isClientPartnerNullable == true;
            var userReference = new EntityReference(Schema.SystemUser, creatingUserId);

            trace?.Trace(
                "TRACE-PLUGIN1: User flags. {0}={1}; {2}={3}.",
                Schema.UserIsClientManager,
                isClientManagerNullable.HasValue ? isClientManagerNullable.Value.ToString() : "null/not returned",
                Schema.UserIsClientPartner,
                isClientPartnerNullable.HasValue ? isClientPartnerNullable.Value.ToString() : "null/not returned");

            DefaultOwnerIfClientManager(target, isClientManager, userReference, trace);
            DefaultClientPartnerIfPartner(target, isClientPartner, userReference, trace);

            if (!isClientManager && !isClientPartner)
            {
                trace?.Trace(
                    "TRACE-PLUGIN1-NOOP: Creating user {0} is not Client Manager or Client Partner.",
                    creatingUserId);
            }
        }

        private static void DefaultOwnerIfClientManager(
            Entity target,
            bool isClientManager,
            EntityReference userReference,
            ITracingService trace)
        {
            if (!isClientManager)
            {
                return;
            }

            if (!HasNonNullAttribute(target, Schema.OpportunityOwner))
            {
                target[Schema.OpportunityOwner] = userReference;

                trace?.Trace(
                    "TRACE-PLUGIN1-ACTION: Defaulted {0} to {1}.",
                    Schema.OpportunityOwner,
                    FormatEntityReference(userReference));
            }
            else
            {
                trace?.Trace(
                    "TRACE-PLUGIN1-NOOP: {0} already supplied. No overwrite.",
                    Schema.OpportunityOwner);
            }
        }

        private static void DefaultClientPartnerIfPartner(
            Entity target,
            bool isClientPartner,
            EntityReference userReference,
            ITracingService trace)
        {
            if (!isClientPartner)
            {
                return;
            }

            if (!HasNonNullAttribute(target, Schema.OpportunityClientPartner))
            {
                target[Schema.OpportunityClientPartner] = userReference;

                trace?.Trace(
                    "TRACE-PLUGIN1-ACTION: Defaulted {0} to {1}.",
                    Schema.OpportunityClientPartner,
                    FormatEntityReference(userReference));
            }
            else
            {
                trace?.Trace(
                    "TRACE-PLUGIN1-NOOP: {0} already supplied. No overwrite.",
                    Schema.OpportunityClientPartner);
            }
        }
    }

    #region Plugin 2 - Architect From Sub-Solution

    /// <summary>
    /// Defaults Opportunity Architect from the selected Sub-Solution.
    ///
    /// Business rules:
    /// 1. When opportunity.ctp_subsolutions is selected/changed,
    ///    copy ctp_subsolutions.ctp_architect to opportunity.new_architectid.
    /// 2. If opportunity.new_architectid is explicitly supplied in the same save,
    ///    preserve the supplied value.
    /// 3. If opportunity.ctp_subsolutions is cleared,
    ///    clear opportunity.new_architectid.
    /// 4. If selected Sub-Solution has no default Architect,
    ///    clear opportunity.new_architectid.
    ///
    /// Register Step A:
    /// - Table: opportunity
    /// - Message: Create
    /// - Stage: PreOperation
    /// - Mode: Synchronous
    /// - Filtering Attributes: none
    /// - Images: none
    ///
    /// Register Step B:
    /// - Table: opportunity
    /// - Message: Update
    /// - Stage: PreOperation
    /// - Mode: Synchronous
    /// - Filtering Attributes: ctp_subsolutions
    /// - Images: none required
    /// </summary>
    public class OpportunitySubSolutionArchitectPlugin : PluginBase
    {
        protected override void ExecuteInternal(
            IPluginExecutionContext context,
            IOrganizationService service,
            ITracingService trace)
        {
            trace?.Trace("TRACE-PLUGIN2: Starting OpportunitySubSolutionArchitectPlugin.");

            if (IsDepthGreaterThanOne(context))
            {
                trace?.Trace("TRACE-PLUGIN2-SKIP: Depth {0} > 1. Exiting to avoid recursion.", context.Depth);
                return;
            }

            if (!IsValidCreateOrUpdateOpportunityPreOperation(context))
            {
                trace?.Trace(
                    "TRACE-PLUGIN2-SKIP: Expected Create/Update opportunity PreOperation. Actual Message={0}, Entity={1}, Stage={2}.",
                    context.MessageName,
                    context.PrimaryEntityName,
                    context.Stage);
                return;
            }

            var target = GetTargetEntity(context);
            TraceTarget(trace, target);

            if (target == null)
            {
                trace?.Trace("TRACE-PLUGIN2-SKIP: Target was not an Entity.");
                return;
            }

            if (!ShouldProcessSubSolutionChange(context, target, trace))
            {
                return;
            }

            if (IsArchitectExplicitlySupplied(target, trace))
            {
                return;
            }

            ApplyArchitectDefaultFromSubSolution(target, service, trace);

            trace?.Trace("TRACE-PLUGIN2: Completed OpportunitySubSolutionArchitectPlugin.");
        }

        private static bool IsValidCreateOrUpdateOpportunityPreOperation(IPluginExecutionContext context)
        {
            return IsPreOperation(context) &&
                   IsPrimaryEntity(context, Schema.Opportunity) &&
                   (IsMessage(context, "Create") || IsMessage(context, "Update"));
        }

        private static bool ShouldProcessSubSolutionChange(
            IPluginExecutionContext context,
            Entity target,
            ITracingService trace)
        {
            if (IsMessage(context, "Update") &&
                !target.Attributes.Contains(Schema.OpportunitySubSolution))
            {
                trace?.Trace(
                    "TRACE-PLUGIN2-SKIP: Update did not include {0}. Target Attributes=[{1}].",
                    Schema.OpportunitySubSolution,
                    FormatKeys(target.Attributes));
                return false;
            }

            if (!target.Attributes.Contains(Schema.OpportunitySubSolution))
            {
                trace?.Trace(
                    "TRACE-PLUGIN2-SKIP: No {0} supplied on Target.",
                    Schema.OpportunitySubSolution);
                return false;
            }

            return true;
        }

        private static bool IsArchitectExplicitlySupplied(Entity target, ITracingService trace)
        {
            if (!target.Attributes.Contains(Schema.OpportunityArchitect))
            {
                return false;
            }

            trace?.Trace(
                "TRACE-PLUGIN2-NOOP: {0} explicitly supplied as {1}. Preserving supplied value.",
                Schema.OpportunityArchitect,
                FormatEntityReference(target.GetAttributeValue<EntityReference>(Schema.OpportunityArchitect)));

            return true;
        }

        private static void ApplyArchitectDefaultFromSubSolution(
            Entity target,
            IOrganizationService service,
            ITracingService trace)
        {
            var subSolutionRef = target.GetAttributeValue<EntityReference>(Schema.OpportunitySubSolution);

            trace?.Trace(
                "TRACE-PLUGIN2: Resolved {0}={1}.",
                Schema.OpportunitySubSolution,
                FormatEntityReference(subSolutionRef));

            if (subSolutionRef == null)
            {
                ClearArchitect(target, trace, "Sub-Solution was cleared");
                return;
            }

            var subSolution = service.Retrieve(
                Schema.SubSolution,
                subSolutionRef.Id,
                new ColumnSet(Schema.SubSolutionDefaultArchitect));

            if (subSolution == null)
            {
                trace?.Trace(
                    "TRACE-PLUGIN2-SKIP: Sub-Solution {0} could not be retrieved.",
                    subSolutionRef.Id);
                return;
            }

            var architectRef = subSolution.GetAttributeValue<EntityReference>(Schema.SubSolutionDefaultArchitect);

            trace?.Trace(
                "TRACE-PLUGIN2: Retrieved {0}={1}.",
                Schema.SubSolutionDefaultArchitect,
                FormatEntityReference(architectRef));

            if (architectRef == null)
            {
                ClearArchitect(target, trace, "Selected Sub-Solution has no default Architect");
                return;
            }

            target[Schema.OpportunityArchitect] = architectRef;

            trace?.Trace(
                "TRACE-PLUGIN2-ACTION: Defaulted {0} to {1} from Sub-Solution {2}.",
                Schema.OpportunityArchitect,
                FormatEntityReference(architectRef),
                subSolutionRef.Id);
        }

        private static void ClearArchitect(Entity target, ITracingService trace, string reason)
        {
            target[Schema.OpportunityArchitect] = null;

            trace?.Trace(
                "TRACE-PLUGIN2-ACTION: Cleared {0}. Reason={1}.",
                Schema.OpportunityArchitect,
                reason);
        }
    }

    #region Plugin 3 - Opportunity Won Standardization

    /// <summary>
    /// Standardizes Opportunity probability when Opportunity is won.
    ///
    /// Supported paths:
    /// 1. Update path:
    ///    PreOperation Update on opportunity when statecode/statuscode changes.
    /// 2. WinOpportunity path:
    ///    PostOperation WinOpportunity because the close request does not provide an opportunity Target to modify in PreOperation.
    /// 3. Optional SetState path:
    ///    PostOperation SetState / SetStateDynamicEntity if the org uses older/custom close behavior.
    ///
    /// Register Step A:
    /// - Table: opportunity
    /// - Message: Update
    /// - Stage: PreOperation
    /// - Mode: Synchronous
    /// - Filtering Attributes: statecode,statuscode
    /// - PreImage Name: PreImage
    /// - PreImage Fields: statecode,statuscode
    ///
    /// Register Step B:
    /// - Table: opportunity
    /// - Message: WinOpportunity
    /// - Stage: PostOperation
    /// - Mode: Synchronous
    /// - Filtering Attributes: none
    /// - Images: none
    ///
    /// Optional Step C:
    /// - Table: opportunity
    /// - Message: SetState or SetStateDynamicEntity
    /// - Stage: PostOperation
    /// - Mode: Synchronous
    /// - Images: none
    /// </summary>
    public class OpportunityWonStandardizationPlugin : PluginBase
    {
        protected override void ExecuteInternal(
            IPluginExecutionContext context,
            IOrganizationService service,
            ITracingService trace)
        {
            trace?.Trace("TRACE-PLUGIN3: Starting OpportunityWonStandardizationPlugin.");

            if (IsDepthGreaterThanOne(context))
            {
                trace?.Trace("TRACE-PLUGIN3-SKIP: Depth {0} > 1. Exiting to avoid recursion.", context.Depth);
                return;
            }

            if (IsMessage(context, "Update"))
            {
                HandleUpdatePath(context, trace);
                return;
            }

            if (IsMessage(context, "WinOpportunity"))
            {
                HandleWinOpportunityPath(context, service, trace);
                return;
            }

            if (IsMessage(context, "SetState") || IsMessage(context, "SetStateDynamicEntity"))
            {
                HandleSetStatePath(context, service, trace);
                return;
            }

            trace?.Trace("TRACE-PLUGIN3-SKIP: Message {0} is not handled.", context.MessageName);
        }

        private static void HandleUpdatePath(
            IPluginExecutionContext context,
            ITracingService trace)
        {
            trace?.Trace("TRACE-PLUGIN3: Handling Update path.");

            if (!IsPreOperation(context) ||
                !IsPrimaryEntity(context, Schema.Opportunity))
            {
                trace?.Trace(
                    "TRACE-PLUGIN3-SKIP: Update path expected opportunity PreOperation. Actual Entity={0}, Stage={1}.",
                    context.PrimaryEntityName,
                    context.Stage);
                return;
            }

            var target = GetTargetEntity(context);
            TraceTarget(trace, target);

            if (target == null)
            {
                trace?.Trace("TRACE-PLUGIN3-SKIP: Target was not an Entity.");
                return;
            }

            var preImage = GetPreImage(context);
            TracePreImage(trace, preImage);

            var resolvedState = GetAttributeValueFromTargetOrPreImage<OptionSetValue>(
                target,
                preImage,
                Schema.OpportunityStateCode);

            trace?.Trace(
                "TRACE-PLUGIN3: Update path resolved statecode={0}.",
                FormatOptionSet(resolvedState));

            if (resolvedState == null || resolvedState.Value != Schema.OpportunityStateWon)
            {
                trace?.Trace(
                    "TRACE-PLUGIN3-NOOP: Opportunity is not Won. Expected={0}; Actual={1}.",
                    Schema.OpportunityStateWon,
                    FormatOptionSet(resolvedState));
                return;
            }

            SetWonProbabilityFields(target);

            trace?.Trace(
                "TRACE-PLUGIN3-ACTION: Update path set {0}=100 and {1}={2}.",
                Schema.OpportunityProbability,
                Schema.OpportunityCustomCloseProbability,
                Schema.CustomCloseProbabilityClosed);
        }

        private static void HandleWinOpportunityPath(
            IPluginExecutionContext context,
            IOrganizationService service,
            ITracingService trace)
        {
            trace?.Trace("TRACE-PLUGIN3: Handling WinOpportunity path.");

            if (!IsPostOperation(context))
            {
                trace?.Trace(
                    "TRACE-PLUGIN3-SKIP: WinOpportunity must be PostOperation stage 40. Actual Stage={0}.",
                    context.Stage);
                return;
            }

            var opportunityRef = GetOpportunityReferenceFromWinOpportunity(context, trace);

            if (opportunityRef == null || opportunityRef.Id == Guid.Empty)
            {
                trace?.Trace("TRACE-PLUGIN3-SKIP: WinOpportunity did not provide a valid Opportunity reference.");
                return;
            }

            UpdateWonProbabilityFields(service, opportunityRef.Id, trace, "WinOpportunity");
        }

        private static EntityReference GetOpportunityReferenceFromWinOpportunity(
            IPluginExecutionContext context,
            ITracingService trace)
        {
            if (context.InputParameters == null ||
                !context.InputParameters.Contains(Schema.OpportunityCloseInputParameter) ||
                !(context.InputParameters[Schema.OpportunityCloseInputParameter] is Entity opportunityClose))
            {
                trace?.Trace("TRACE-PLUGIN3-SKIP: WinOpportunity did not include OpportunityClose Entity.");
                return null;
            }

            trace?.Trace(
                "TRACE-PLUGIN3: OpportunityClose found. LogicalName={0}; Attributes=[{1}].",
                opportunityClose.LogicalName,
                FormatKeys(opportunityClose.Attributes));

            var opportunityRef = opportunityClose.GetAttributeValue<EntityReference>(Schema.OpportunityCloseOpportunity);

            trace?.Trace(
                "TRACE-PLUGIN3: WinOpportunity Opportunity reference={0}.",
                FormatEntityReference(opportunityRef));

            return opportunityRef;
        }

        private static void HandleSetStatePath(
            IPluginExecutionContext context,
            IOrganizationService service,
            ITracingService trace)
        {
            trace?.Trace("TRACE-PLUGIN3: Handling SetState/SetStateDynamicEntity path.");

            if (!IsPostOperation(context))
            {
                trace?.Trace(
                    "TRACE-PLUGIN3-SKIP: SetState must be PostOperation stage 40. Actual Stage={0}.",
                    context.Stage);
                return;
            }

            var entityMoniker = GetInputParameter<EntityReference>(context, Schema.EntityMonikerInputParameter);
            var state = GetInputParameter<OptionSetValue>(context, Schema.StateInputParameter);

            trace?.Trace(
                "TRACE-PLUGIN3: SetState inputs. EntityMoniker={0}; State={1}.",
                FormatEntityReference(entityMoniker),
                FormatOptionSet(state));

            if (entityMoniker == null ||
                entityMoniker.Id == Guid.Empty ||
                !string.Equals(entityMoniker.LogicalName, Schema.Opportunity, StringComparison.OrdinalIgnoreCase))
            {
                trace?.Trace("TRACE-PLUGIN3-SKIP: EntityMoniker is not a valid Opportunity reference.");
                return;
            }

            if (state == null || state.Value != Schema.OpportunityStateWon)
            {
                trace?.Trace(
                    "TRACE-PLUGIN3-NOOP: SetState state is not Won. Expected={0}; Actual={1}.",
                    Schema.OpportunityStateWon,
                    FormatOptionSet(state));
                return;
            }

            UpdateWonProbabilityFields(service, entityMoniker.Id, trace, "SetState");
        }

        private static T GetInputParameter<T>(IPluginExecutionContext context, string parameterName) where T : class
        {
            if (context.InputParameters != null &&
                context.InputParameters.Contains(parameterName) &&
                context.InputParameters[parameterName] is T value)
            {
                return value;
            }

            return null;
        }

        private static void UpdateWonProbabilityFields(
            IOrganizationService service,
            Guid opportunityId,
            ITracingService trace,
            string sourcePath)
        {
            var update = new Entity(Schema.Opportunity, opportunityId);
            SetWonProbabilityFields(update);

            service.Update(update);

            trace?.Trace(
                "TRACE-PLUGIN3-ACTION: {0} path updated opportunity {1}. Set {2}=100 and {3}={4}.",
                sourcePath,
                opportunityId,
                Schema.OpportunityProbability,
                Schema.OpportunityCustomCloseProbability,
                Schema.CustomCloseProbabilityClosed);
        }
