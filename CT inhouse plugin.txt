using System;
using Microsoft.Xrm.Sdk;
using Microsoft.Xrm.Sdk.Query;

namespace Ctp.Crm.OpportunityAutomation
{
    /// <summary>
    /// Production schema constants validated against the uploaded metadata:
    /// - Opportunity: opportunity
    /// - User: systemuser
    /// - Sub-Solution: ctp_subsolutions
    /// </summary>
    public static class Schema
    {
        // Tables
        public const string Opportunity = "opportunity";
        public const string SystemUser = "systemuser";
        public const string SubSolution = "ctp_subsolutions";

        // Opportunity fields
        public const string OpportunityOwner = "ownerid";                       // Display name: Client Manager
        public const string OpportunityClientPartner = "new_clientpartnerid";   // Lookup to systemuser
        public const string OpportunityArchitect = "new_architectid";           // Lookup to systemuser
        public const string OpportunitySubSolution = "ctp_subsolutions";        // Lookup to ctp_subsolutions
        public const string OpportunityProbability = "closeprobability";        // OOB whole number probability
        public const string OpportunityCustomCloseProbability = "new_closeprobability"; // Custom choice field
        public const string OpportunityStateCode = "statecode";
        public const string OpportunityStatusCode = "statuscode";

        // User fields
        public const string UserIsClientManager = "new_clientmanager";          // Two options
        public const string UserIsClientPartner = "new_clientpartner";          // Two options

        // Sub-Solution fields
        public const string SubSolutionDefaultArchitect = "ctp_architect";      // Lookup to systemuser

        // Dataverse state values for Opportunity
        public const int OpportunityStateOpen = 0;
        public const int OpportunityStateWon = 1;
        public const int OpportunityStateLost = 2;

        // new_closeprobability option value from uploaded metadata
        public const int CustomCloseProbabilityClosed = 100000008;
    }

    /// <summary>
    /// Shared base class for safe tracing, target extraction, and image helpers.
    /// Register plugin steps in sandbox, synchronous, PreOperation unless otherwise noted.
    /// </summary>
    public abstract class PluginBase : IPlugin
    {
        protected const int PreOperationStage = 20;

        public void Execute(IServiceProvider serviceProvider)
        {
            if (serviceProvider == null)
            {
                throw new InvalidPluginExecutionException("Service provider is null.");
            }

            var context = (IPluginExecutionContext)serviceProvider.GetService(typeof(IPluginExecutionContext));
            var trace = (ITracingService)serviceProvider.GetService(typeof(ITracingService));
            var serviceFactory = (IOrganizationServiceFactory)serviceProvider.GetService(typeof(IOrganizationServiceFactory));

            if (context == null)
            {
                throw new InvalidPluginExecutionException("Plugin execution context is null.");
            }

            if (serviceFactory == null)
            {
                throw new InvalidPluginExecutionException("Organization service factory is null.");
            }

            // Uses SYSTEM context for reads so defaulting logic is not blocked by end-user read privileges.
            // Field changes are applied only through Target in the current transaction.
            var service = serviceFactory.CreateOrganizationService(null);

            try
            {
                trace?.Trace(
                    "Entered {0}. Message={1}, Entity={2}, Stage={3}, Mode={4}, Depth={5}, UserId={6}, InitiatingUserId={7}",
                    GetType().FullName,
                    context.MessageName,
                    context.PrimaryEntityName,
                    context.Stage,
                    context.Mode,
                    context.Depth,
                    context.UserId,
                    context.InitiatingUserId);

                ExecuteInternal(context, service, trace);

                trace?.Trace("Exited {0} successfully.", GetType().FullName);
            }
            catch (InvalidPluginExecutionException)
            {
                throw;
            }
            catch (Exception ex)
            {
                trace?.Trace("Unhandled exception in {0}: {1}", GetType().FullName, ex);
                throw new InvalidPluginExecutionException($"An error occurred in {GetType().Name}: {ex.Message}", ex);
            }
        }

        protected abstract void ExecuteInternal(
            IPluginExecutionContext context,
            IOrganizationService service,
            ITracingService trace);

        protected static Entity GetTargetEntity(IPluginExecutionContext context)
        {
            if (context.InputParameters != null &&
                context.InputParameters.Contains("Target") &&
                context.InputParameters["Target"] is Entity target)
            {
                return target;
            }

            return null;
        }

        protected static Entity GetPreImage(IPluginExecutionContext context, string imageName = "PreImage")
        {
            if (context.PreEntityImages != null && context.PreEntityImages.Contains(imageName))
            {
                return context.PreEntityImages[imageName];
            }

            return null;
        }

        protected static T GetAttributeValueFromTargetOrPreImage<T>(
            Entity target,
            Entity preImage,
            string attributeName)
        {
            if (target != null && target.Attributes.Contains(attributeName))
            {
                return target.GetAttributeValue<T>(attributeName);
            }

            if (preImage != null && preImage.Attributes.Contains(attributeName))
            {
                return preImage.GetAttributeValue<T>(attributeName);
            }

            return default(T);
        }

        protected static bool IsMessage(IPluginExecutionContext context, string messageName)
        {
            return string.Equals(context.MessageName, messageName, StringComparison.OrdinalIgnoreCase);
        }

        protected static bool IsPrimaryEntity(IPluginExecutionContext context, string entityName)
        {
            return string.Equals(context.PrimaryEntityName, entityName, StringComparison.OrdinalIgnoreCase);
        }

        protected static bool IsPreOperation(IPluginExecutionContext context)
        {
            return context.Stage == PreOperationStage;
        }
    }

    /// <summary>
    /// Plugin 1: Defaults Opportunity ownership fields during Opportunity Create.
    ///
    /// Business rules:
    /// - If creating user has systemuser.new_clientmanager = true,
    ///   default opportunity.ownerid to that user when ownerid was not already supplied.
    ///
    /// - If creating user has systemuser.new_clientpartner = true,
    ///   default opportunity.new_clientpartnerid when it was not already supplied.
    ///
    /// - If user has both flags, both fields are defaulted if blank.
    ///
    /// Register:
    /// - Table: opportunity
    /// - Message: Create
    /// - Stage: PreOperation
    /// - Mode: Synchronous
    /// - Filtering Attributes: none
    /// - Images: none
    /// </summary>
    public class OpportunityOwnershipDefaultsPlugin : PluginBase
    {
        protected override void ExecuteInternal(
            IPluginExecutionContext context,
            IOrganizationService service,
            ITracingService trace)
        {
            if (!IsPreOperation(context) ||
                !IsMessage(context, "Create") ||
                !IsPrimaryEntity(context, Schema.Opportunity))
            {
                return;
            }

            var target = GetTargetEntity(context);
            if (target == null)
            {
                trace?.Trace("Target was not an Entity. Exiting.");
                return;
            }

            var creatingUserId = context.InitiatingUserId != Guid.Empty
                ? context.InitiatingUserId
                : context.UserId;

            if (creatingUserId == Guid.Empty)
            {
                trace?.Trace("Creating user could not be resolved. Exiting.");
                return;
            }

            var user = service.Retrieve(
                Schema.SystemUser,
                creatingUserId,
                new ColumnSet(
                    Schema.UserIsClientManager,
                    Schema.UserIsClientPartner));

            if (user == null)
            {
                trace?.Trace("System user {0} could not be retrieved. Exiting.", creatingUserId);
                return;
            }

            var isClientManager = user.GetAttributeValue<bool>(Schema.UserIsClientManager);
            var isClientPartner = user.GetAttributeValue<bool>(Schema.UserIsClientPartner);
            var userReference = new EntityReference(Schema.SystemUser, creatingUserId);

            if (isClientManager)
            {
                if (!target.Attributes.Contains(Schema.OpportunityOwner) ||
                    target[Schema.OpportunityOwner] == null)
                {
                    target[Schema.OpportunityOwner] = userReference;
                    trace?.Trace("Defaulted {0} to creating user {1}.", Schema.OpportunityOwner, creatingUserId);
                }
                else
                {
                    trace?.Trace("{0} was already supplied. No overwrite.", Schema.OpportunityOwner);
                }
            }

            if (isClientPartner)
            {
                if (!target.Attributes.Contains(Schema.OpportunityClientPartner) ||
                    target[Schema.OpportunityClientPartner] == null)
                {
                    target[Schema.OpportunityClientPartner] = userReference;
                    trace?.Trace("Defaulted {0} to creating user {1}.", Schema.OpportunityClientPartner, creatingUserId);
                }
                else
                {
                    trace?.Trace("{0} was already supplied. No overwrite.", Schema.OpportunityClientPartner);
                }
            }

            if (!isClientManager && !isClientPartner)
            {
                trace?.Trace(
                    "Creating user {0} is not marked as Client Manager or Client Partner. No defaults applied.",
                    creatingUserId);
            }
        }
    }

    /// <summary>
    /// Plugin 2: Defaults Opportunity Architect from selected Sub-Solution.
    ///
    /// Business rule:
    /// - When opportunity.ctp_subsolutions is selected or changed,
    ///   retrieve ctp_subsolutions.ctp_architect and populate opportunity.new_architectid.
    ///
    /// - If new_architectid is explicitly supplied in the same save operation,
    ///   the plugin preserves that value.
    ///
    /// - If ctp_subsolutions is cleared and new_architectid is not explicitly supplied,
    ///   the plugin clears new_architectid.
    ///
    /// Register Step A:
    /// - Table: opportunity
    /// - Message: Create
    /// - Stage: PreOperation
    /// - Mode: Synchronous
    /// - Filtering Attributes: none
    /// - Images: none
    ///
    /// Register Step B:
    /// - Table: opportunity
    /// - Message: Update
    /// - Stage: PreOperation
    /// - Mode: Synchronous
    /// - Filtering Attributes: ctp_subsolutions
    /// - Images: none required
    /// </summary>
    public class OpportunitySubSolutionArchitectPlugin : PluginBase
    {
        protected override void ExecuteInternal(
            IPluginExecutionContext context,
            IOrganizationService service,
            ITracingService trace)
        {
            if (!IsPreOperation(context) ||
                !IsPrimaryEntity(context, Schema.Opportunity))
            {
                return;
            }

            if (!IsMessage(context, "Create") &&
                !IsMessage(context, "Update"))
            {
                return;
            }

            var target = GetTargetEntity(context);
            if (target == null)
            {
                trace?.Trace("Target was not an Entity. Exiting.");
                return;
            }

            // On update, only run when Sub-Solution is actually part of the update payload.
            // This prevents unrelated opportunity updates from re-defaulting the Architect.
            if (IsMessage(context, "Update") &&
                !target.Attributes.Contains(Schema.OpportunitySubSolution))
            {
                trace?.Trace("Update did not include {0}. Exiting.", Schema.OpportunitySubSolution);
                return;
            }

            // Preserve an Architect explicitly supplied by user/process in the same operation.
            if (target.Attributes.Contains(Schema.OpportunityArchitect))
            {
                trace?.Trace(
                    "{0} was explicitly supplied in Target. Preserving supplied value.",
                    Schema.OpportunityArchitect);

                return;
            }

            if (!target.Attributes.Contains(Schema.OpportunitySubSolution))
            {
                trace?.Trace("No {0} supplied. Exiting.", Schema.OpportunitySubSolution);
                return;
            }

            var subSolutionRef = target.GetAttributeValue<EntityReference>(Schema.OpportunitySubSolution);

            if (subSolutionRef == null)
            {
                target[Schema.OpportunityArchitect] = null;

                trace?.Trace(
                    "{0} was cleared, so {1} was also cleared.",
                    Schema.OpportunitySubSolution,
                    Schema.OpportunityArchitect);

                return;
            }

            var subSolution = service.Retrieve(
                Schema.SubSolution,
                subSolutionRef.Id,
                new ColumnSet(Schema.SubSolutionDefaultArchitect));

            if (subSolution == null)
            {
                trace?.Trace(
                    "Sub-Solution {0} could not be retrieved. Architect not defaulted.",
                    subSolutionRef.Id);

                return;
            }

            var architectRef = subSolution.GetAttributeValue<EntityReference>(
                Schema.SubSolutionDefaultArchitect);

            if (architectRef == null)
            {
                trace?.Trace(
                    "Sub-Solution {0} has no {1}. Architect not defaulted.",
                    subSolutionRef.Id,
                    Schema.SubSolutionDefaultArchitect);

                return;
            }

            target[Schema.OpportunityArchitect] = architectRef;

            trace?.Trace(
                "Defaulted {0} from Sub-Solution {1} field {2}. Architect={3}",
                Schema.OpportunityArchitect,
                subSolutionRef.Id,
                Schema.SubSolutionDefaultArchitect,
                architectRef.Id);
        }
    }

    /// <summary>
    /// Plugin 3: Standardizes Opportunity probability when Opportunity is won.
    ///
    /// Business rule:
    /// - When opportunity.statecode becomes Won, set:
    ///   * closeprobability = 100
    ///   * new_closeprobability = Closed option value 100000008
    ///
    /// Register:
    /// - Table: opportunity
    /// - Message: Update
    /// - Stage: PreOperation
    /// - Mode: Synchronous
    /// - Filtering Attributes: statecode,statuscode
    /// - PreImage: "PreImage" with statecode,statuscode recommended
    /// </summary>
    public class OpportunityWonStandardizationPlugin : PluginBase
    {
        protected override void ExecuteInternal(
            IPluginExecutionContext context,
            IOrganizationService service,
            ITracingService trace)
        {
            if (!IsPreOperation(context) ||
                !IsMessage(context, "Update") ||
                !IsPrimaryEntity(context, Schema.Opportunity))
            {
                return;
            }

            var target = GetTargetEntity(context);
            if (target == null)
            {
                trace?.Trace("Target was not an Entity. Exiting.");
                return;
            }

            var preImage = GetPreImage(context);

            var state = GetAttributeValueFromTargetOrPreImage<OptionSetValue>(
                target,
                preImage,
                Schema.OpportunityStateCode)?.Value;

            if (state != Schema.OpportunityStateWon)
            {
                trace?.Trace(
                    "Opportunity is not Won. Current/resolved statecode={0}. Exiting.",
                    state.HasValue ? state.Value.ToString() : "null");

                return;
            }

            target[Schema.OpportunityProbability] = 100;

            target[Schema.OpportunityCustomCloseProbability] =
                new OptionSetValue(Schema.CustomCloseProbabilityClosed);

            trace?.Trace(
                "Opportunity is Won. Set {0}=100 and {1}={2}.",
                Schema.OpportunityProbability,
                Schema.OpportunityCustomCloseProbability,
                Schema.CustomCloseProbabilityClosed);
        }
    }
}
