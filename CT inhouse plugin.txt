    /// <summary>

    /// </summary>

        // Opportunity fields

        public const string OpportunityClientPartner = "new_clientpartnerid";     // Lookup to systemuser

        public const string OpportunitySubSolution = "ctp_subsolutions";          // Lookup to ctp_subsolutions
        public const string OpportunityProbability = "closeprobability";          // OOB whole number probability
        public const string OpportunityCustomCloseProbability = "new_closeprobability";
        public const string OpportunityStateCode = "statecode";
        public const string OpportunityStatusCode = "statuscode";

        // User fields
        public const string UserIsClientManager = "new_clientmanager";            // Two Options
        public const string UserIsClientPartner = "new_clientpartner";            // Two Options

        // Sub-Solution fields

        // OpportunityClose fields / close request input parameter names

        public const string OpportunityCloseOpportunity = "opportunityid";
        public const string OpportunityCloseInputParameter = "OpportunityClose";

        // Dataverse pipeline stages

        // Dataverse opportunity state values
        public const int OpportunityStateOpen = 0;

    /// <summary>
    /// Shared base class for tracing, target extraction, and image helpers.
    /// </summary>
    public abstract class PluginBase : IPlugin
    {
        public void Execute(IServiceProvider serviceProvider)
        {
            if (serviceProvider == null)
            {
                throw new InvalidPluginExecutionException("Service provider is null.");
            }

            var context = (IPluginExecutionContext)serviceProvider.GetService(typeof(IPluginExecutionContext));
            var trace = (ITracingService)serviceProvider.GetService(typeof(ITracingService));
            var serviceFactory = (IOrganizationServiceFactory)serviceProvider.GetService(typeof(IOrganizationServiceFactory));

            if (context == null)
            {
                throw new InvalidPluginExecutionException("Plugin execution context is null.");
            }

            if (serviceFactory == null)
            {
                throw new InvalidPluginExecutionException("Organization service factory is null.");
            }

            // System context is used for reads/defaulting so user read privileges do not block automation.
            // Write operations are only used where required, for example PostOperation WinOpportunity.
            var service = serviceFactory.CreateOrganizationService(null);

            try
            {
                TraceExecutionStart(context, trace, GetType().FullName);
                ExecuteInternal(context, service, trace);
                trace?.Trace("TRACE-END: {0} completed successfully. CorrelationId={1}", GetType().FullName, context.CorrelationId);
            }
            catch (InvalidPluginExecutionException ex)
            {
                trace?.Trace(
                    "TRACE-ERROR: {0} threw InvalidPluginExecutionException. Message={1}. CorrelationId={2}",
                    GetType().FullName,
                    ex.Message,
                    context.CorrelationId);
                throw;
            }
            catch (Exception ex)
            {
                trace?.Trace(
                    "TRACE-ERROR: Unhandled exception in {0}. CorrelationId={1}. Exception={2}",
                    GetType().FullName,
                    context.CorrelationId,
                    ex);

                throw new InvalidPluginExecutionException($"An error occurred in {GetType().Name}: {ex.Message}", ex);
            }
        }

        protected abstract void ExecuteInternal(
            IPluginExecutionContext context,
            IOrganizationService service,
            ITracingService trace);

        protected static Entity GetTargetEntity(IPluginExecutionContext context)
        {
            if (context.InputParameters != null &&
                context.InputParameters.Contains("Target") &&
                context.InputParameters["Target"] is Entity target)
            {
                return target;
            }

            return null;
        }

        protected static Entity GetPreImage(IPluginExecutionContext context, string imageName = "PreImage")
        {
            if (context.PreEntityImages != null && context.PreEntityImages.Contains(imageName))
            {
                return context.PreEntityImages[imageName];
            }

            return null;
        }

        protected static T GetAttributeValueFromTargetOrPreImage<T>(
            Entity target,
            Entity preImage,
            string attributeName)
        {
            if (target != null && target.Attributes.Contains(attributeName))
            {
                return target.GetAttributeValue<T>(attributeName);
            }

            if (preImage != null && preImage.Attributes.Contains(attributeName))
            {
                return preImage.GetAttributeValue<T>(attributeName);
            }

            return default(T);
        }

        protected static bool IsMessage(IPluginExecutionContext context, string messageName)
        {
            return string.Equals(context.MessageName, messageName, StringComparison.OrdinalIgnoreCase);
        }

        protected static bool IsPrimaryEntity(IPluginExecutionContext context, string entityName)
        {
            return string.Equals(context.PrimaryEntityName, entityName, StringComparison.OrdinalIgnoreCase);
        }

        protected static bool IsPreOperation(IPluginExecutionContext context)
        {
            return context.Stage == Schema.PreOperationStage;
        }

        protected static bool IsPostOperation(IPluginExecutionContext context)
        {
            return context.Stage == Schema.PostOperationStage;
        }

        protected static bool? GetNullableBoolean(Entity entity, string attributeName)
        {
            if (entity == null || !entity.Attributes.Contains(attributeName) || entity[attributeName] == null)
            {
                return null;
            }

            return entity.GetAttributeValue<bool>(attributeName);
        }

        protected static void SetWonProbabilityFields(Entity opportunity)
        {
            opportunity[Schema.OpportunityProbability] = 100;
            opportunity[Schema.OpportunityCustomCloseProbability] =
                new OptionSetValue(Schema.CustomCloseProbabilityClosed);
        }

        protected static void TraceExecutionStart(
            IPluginExecutionContext context,
            ITracingService trace,
            string pluginName)
        {
            trace?.Trace("TRACE-START: {0}", pluginName);

            trace?.Trace(
                "TRACE-CONTEXT: Message={0}; PrimaryEntity={1}; PrimaryEntityId={2}; Stage={3}; Mode={4}; Depth={5}; UserId={6}; InitiatingUserId={7}; OrganizationId={8}; CorrelationId={9}; OperationId={10}",
                context.MessageName,
                context.PrimaryEntityName,
                context.PrimaryEntityId,
                context.Stage,
                context.Mode,
                context.Depth,
                context.UserId,
                context.InitiatingUserId,
                context.OrganizationId,
                context.CorrelationId,
                context.OperationId);

            trace?.Trace(
                "TRACE-PARAMETERS: InputParameters=[{0}]; OutputParameters=[{1}]; SharedVariables=[{2}]",
                FormatKeys(context.InputParameters),
                FormatKeys(context.OutputParameters),
                FormatKeys(context.SharedVariables));
        }

        protected static void TraceTarget(ITracingService trace, Entity target)
        {
            if (target == null)
            {
                trace?.Trace("TRACE-TARGET: Target is null.");
                return;
            }

            trace?.Trace(
                "TRACE-TARGET: LogicalName={0}; Id={1}; AttributeCount={2}; Attributes=[{3}]",
                target.LogicalName,
                target.Id,
                target.Attributes.Count,
                FormatKeys(target.Attributes));
        }

        protected static void TracePreImage(
            ITracingService trace,
            Entity preImage,
            string imageName = "PreImage")
        {
            if (preImage == null)
            {
                trace?.Trace("TRACE-PREIMAGE: {0} was not available.", imageName);
                return;
            }

            trace?.Trace(
                "TRACE-PREIMAGE: {0}; LogicalName={1}; Id={2}; AttributeCount={3}; Attributes=[{4}]",
                imageName,
                preImage.LogicalName,
                preImage.Id,
                preImage.Attributes.Count,
                FormatKeys(preImage.Attributes));
        }

        protected static string FormatKeys(ParameterCollection parameters)
        {
            if (parameters == null || parameters.Count == 0)
            {
                return "none";
            }

            return string.Join(", ", parameters.Keys);
        }

        protected static string FormatKeys(AttributeCollection attributes)
        {
            if (attributes == null || attributes.Count == 0)
            {
                return "none";
            }

            return string.Join(", ", attributes.Keys);
        }

        protected static string FormatEntityReference(EntityReference entityReference)
        {
            if (entityReference == null)
            {
                return "null";
            }

            return $"{entityReference.LogicalName}:{entityReference.Id}";
        }

        protected static string FormatOptionSet(OptionSetValue optionSetValue)
        {
            return optionSetValue == null ? "null" : optionSetValue.Value.ToString();
        }
    }

    /// <summary>
    /// Plugin 1: Defaults Opportunity ownership fields during Opportunity Create.
    ///
    /// Business rules:
    /// - If creating user has systemuser.new_clientmanager = true, default opportunity.ownerid.
    /// - If creating user has systemuser.new_clientpartner = true, default opportunity.new_clientpartnerid.
    /// - If user has both flags, both fields are defaulted if blank.
    ///
    /// Register:
    /// - Table: opportunity
    /// - Message: Create
    /// - Stage: PreOperation
    /// - Mode: Synchronous
    /// - Filtering Attributes: none
    /// - Images: none
    /// </summary>
    public class OpportunityOwnershipDefaultsPlugin : PluginBase
    {
        protected override void ExecuteInternal(
            IPluginExecutionContext context,
            IOrganizationService service,
            ITracingService trace)
        {
            trace?.Trace("TRACE-PLUGIN1: Starting OpportunityOwnershipDefaultsPlugin.");

            if (context.Depth > 1)
            {
                trace?.Trace("TRACE-PLUGIN1-SKIP: Depth {0} > 1. Exiting to avoid recursion.", context.Depth);
                return;
            }

            if (!IsPreOperation(context) ||
                !IsMessage(context, "Create") ||
                !IsPrimaryEntity(context, Schema.Opportunity))
            {
                trace?.Trace(
                    "TRACE-PLUGIN1-SKIP: Step did not match. Expected Message=Create, Entity=opportunity, Stage=20. Actual Message={0}, Entity={1}, Stage={2}.",
                    context.MessageName,
                    context.PrimaryEntityName,
                    context.Stage);
                return;
            }

            var target = GetTargetEntity(context);
            TraceTarget(trace, target);

            if (target == null)
            {
                trace?.Trace("TRACE-PLUGIN1-SKIP: Target was not an Entity.");
                return;
            }

            var creatingUserId = context.InitiatingUserId != Guid.Empty
                ? context.InitiatingUserId
                : context.UserId;

            trace?.Trace(
                "TRACE-PLUGIN1: Resolved creating user. InitiatingUserId={0}; UserId={1}; CreatingUserId={2}",
                context.InitiatingUserId,
                context.UserId,
                creatingUserId);

            if (creatingUserId == Guid.Empty)
            {
                trace?.Trace("TRACE-PLUGIN1-SKIP: Creating user could not be resolved.");
                return;
            }

            trace?.Trace(
                "TRACE-PLUGIN1: Retrieving systemuser {0} with columns [{1}, {2}].",
                creatingUserId,
                Schema.UserIsClientManager,
                Schema.UserIsClientPartner);

            var user = service.Retrieve(
                Schema.SystemUser,
                creatingUserId,
                new ColumnSet(
                    Schema.UserIsClientManager,
                    Schema.UserIsClientPartner));

            if (user == null)
            {
                trace?.Trace("TRACE-PLUGIN1-SKIP: System user {0} could not be retrieved.", creatingUserId);
                return;
            }

            bool? isClientManagerNullable = GetNullableBoolean(user, Schema.UserIsClientManager);
            bool? isClientPartnerNullable = GetNullableBoolean(user, Schema.UserIsClientPartner);

            bool isClientManager = isClientManagerNullable == true;
            bool isClientPartner = isClientPartnerNullable == true;

            var userReference = new EntityReference(Schema.SystemUser, creatingUserId);

            trace?.Trace(
                "TRACE-PLUGIN1: User flags loaded. {0}={1}; {2}={3}.",
                Schema.UserIsClientManager,
                isClientManagerNullable.HasValue ? isClientManagerNullable.Value.ToString() : "null/not returned",
                Schema.UserIsClientPartner,
                isClientPartnerNullable.HasValue ? isClientPartnerNullable.Value.ToString() : "null/not returned");

            if (isClientManager)
            {
                var ownerAlreadySupplied =
                    target.Attributes.Contains(Schema.OpportunityOwner) &&
                    target[Schema.OpportunityOwner] != null;

                trace?.Trace(
                    "TRACE-PLUGIN1: Client Manager rule matched. Target already has {0}={1}.",
                    Schema.OpportunityOwner,
                    ownerAlreadySupplied
                        ? FormatEntityReference(target.GetAttributeValue<EntityReference>(Schema.OpportunityOwner))
                        : "null/not supplied");

                if (!ownerAlreadySupplied)
                {
                    target[Schema.OpportunityOwner] = userReference;

                    trace?.Trace(
                        "TRACE-PLUGIN1-ACTION: Defaulted {0} to {1}.",
                        Schema.OpportunityOwner,
                        FormatEntityReference(userReference));
                }
                else
                {
                    trace?.Trace(
                        "TRACE-PLUGIN1-NOOP: {0} was already supplied. No overwrite.",
                        Schema.OpportunityOwner);
                }
            }

            if (isClientPartner)
            {
                var partnerAlreadySupplied =
                    target.Attributes.Contains(Schema.OpportunityClientPartner) &&
                    target[Schema.OpportunityClientPartner] != null;

                trace?.Trace(
                    "TRACE-PLUGIN1: Client Partner rule matched. Target already has {0}={1}.",
                    Schema.OpportunityClientPartner,
                    partnerAlreadySupplied
                        ? FormatEntityReference(target.GetAttributeValue<EntityReference>(Schema.OpportunityClientPartner))
                        : "null/not supplied");

                if (!partnerAlreadySupplied)
                {
                    target[Schema.OpportunityClientPartner] = userReference;

                    trace?.Trace(
                        "TRACE-PLUGIN1-ACTION: Defaulted {0} to {1}.",
                        Schema.OpportunityClientPartner,
                        FormatEntityReference(userReference));
                }
                else
                {
                    trace?.Trace(
                        "TRACE-PLUGIN1-NOOP: {0} was already supplied. No overwrite.",
                        Schema.OpportunityClientPartner);
                }
            }

            if (!isClientManager && !isClientPartner)
            {
                trace?.Trace(
                    "TRACE-PLUGIN1-NOOP: Creating user {0} is not marked as Client Manager or Client Partner.",
                    creatingUserId);
            }

            trace?.Trace("TRACE-PLUGIN1: Completed OpportunityOwnershipDefaultsPlugin.");
        }
    }

    /// <summary>
    /// Plugin 2: Defaults the Opportunity Architect from the selected Sub-Solution.
    ///
    /// Business rules:
    /// - When opportunity.ctp_subsolutions is selected/changed, copy ctp_subsolutions.ctp_architect to opportunity.new_architectid.
    /// - If new_architectid is explicitly supplied in the same save, preserve the supplied value.
    /// - If ctp_subsolutions is cleared, clear new_architectid.
    /// - If selected Sub-Solution has no default architect, clear new_architectid.
    ///
    /// Register Step A:
    /// - Table: opportunity
    /// - Message: Create
    /// - Stage: PreOperation
    /// - Mode: Synchronous
    /// - Filtering Attributes: none
    /// - Images: none
    ///
    /// Register Step B:
    /// - Table: opportunity
    /// - Message: Update
    /// - Stage: PreOperation
    /// - Mode: Synchronous
    /// - Filtering Attributes: ctp_subsolutions
    /// - Images: none required
    /// </summary>
    public class OpportunitySubSolutionArchitectPlugin : PluginBase
    {
        protected override void ExecuteInternal(
            IPluginExecutionContext context,
            IOrganizationService service,
            ITracingService trace)
        {
            trace?.Trace("TRACE-PLUGIN2: Starting OpportunitySubSolutionArchitectPlugin.");

            if (context.Depth > 1)
            {
                trace?.Trace("TRACE-PLUGIN2-SKIP: Depth {0} > 1. Exiting to avoid recursion.", context.Depth);
                return;
            }

            if (!IsPreOperation(context) ||
                !IsPrimaryEntity(context, Schema.Opportunity))
            {
                trace?.Trace(
                    "TRACE-PLUGIN2-SKIP: Step did not match. Expected Entity=opportunity, Stage=20. Actual Entity={0}, Stage={1}.",
                    context.PrimaryEntityName,
                    context.Stage);
                return;
            }

            if (!IsMessage(context, "Create") &&
                !IsMessage(context, "Update"))
            {
                trace?.Trace(
                    "TRACE-PLUGIN2-SKIP: Message {0} is not handled.",
                    context.MessageName);
                return;
            }

            var target = GetTargetEntity(context);
            TraceTarget(trace, target);

            if (target == null)
            {
                trace?.Trace("TRACE-PLUGIN2-SKIP: Target was not an Entity.");
                return;
            }

            if (IsMessage(context, "Update") &&
                !target.Attributes.Contains(Schema.OpportunitySubSolution))
            {
                trace?.Trace(
                    "TRACE-PLUGIN2-SKIP: Update did not include {0}. Current Target attributes=[{1}].",
                    Schema.OpportunitySubSolution,
                    FormatKeys(target.Attributes));
                return;
            }

            if (target.Attributes.Contains(Schema.OpportunityArchitect))
            {
                trace?.Trace(
                    "TRACE-PLUGIN2-NOOP: {0} was explicitly supplied in Target as {1}. Preserving supplied value.",
                    Schema.OpportunityArchitect,
                    FormatEntityReference(target.GetAttributeValue<EntityReference>(Schema.OpportunityArchitect)));
                return;
            }

            if (!target.Attributes.Contains(Schema.OpportunitySubSolution))
            {
                trace?.Trace(
                    "TRACE-PLUGIN2-SKIP: No {0} supplied on Target.",
                    Schema.OpportunitySubSolution);
                return;
            }

            var subSolutionRef = target.GetAttributeValue<EntityReference>(
                Schema.OpportunitySubSolution);

            trace?.Trace(
                "TRACE-PLUGIN2: Resolved Target {0}={1}.",
                Schema.OpportunitySubSolution,
                FormatEntityReference(subSolutionRef));

            if (subSolutionRef == null)
            {
                target[Schema.OpportunityArchitect] = null;

                trace?.Trace(
                    "TRACE-PLUGIN2-ACTION: {0} was cleared, so {1} was also cleared.",
                    Schema.OpportunitySubSolution,
                    Schema.OpportunityArchitect);
                return;
            }

            trace?.Trace(
                "TRACE-PLUGIN2: Retrieving {0} {1} with column [{2}].",
                Schema.SubSolution,
                subSolutionRef.Id,
                Schema.SubSolutionDefaultArchitect);

            var subSolution = service.Retrieve(
                Schema.SubSolution,
                subSolutionRef.Id,
                new ColumnSet(Schema.SubSolutionDefaultArchitect));

            if (subSolution == null)
            {
                trace?.Trace(
                    "TRACE-PLUGIN2-SKIP: Sub-Solution {0} could not be retrieved.",
                    subSolutionRef.Id);
                return;
            }

            var architectRef = subSolution.GetAttributeValue<EntityReference>(
                Schema.SubSolutionDefaultArchitect);

            trace?.Trace(
                "TRACE-PLUGIN2: Retrieved Sub-Solution default architect. {0}={1}.",
                Schema.SubSolutionDefaultArchitect,
                FormatEntityReference(architectRef));

            if (architectRef == null)
            {
                target[Schema.OpportunityArchitect] = null;

                trace?.Trace(
                    "TRACE-PLUGIN2-ACTION: Sub-Solution {0} has no {1}. Cleared {2}.",
                    subSolutionRef.Id,
                    Schema.SubSolutionDefaultArchitect,
                    Schema.OpportunityArchitect);
                return;
            }

            target[Schema.OpportunityArchitect] = architectRef;

            trace?.Trace(
                "TRACE-PLUGIN2-ACTION: Defaulted {0} to {1} from {2} {3}.",
                Schema.OpportunityArchitect,
                FormatEntityReference(architectRef),
                Schema.SubSolution,
                subSolutionRef.Id);

            trace?.Trace("TRACE-PLUGIN2: Completed OpportunitySubSolutionArchitectPlugin.");
        }
    }

    /// <summary>
    /// Plugin 3: Standardizes Opportunity probability when Opportunity is won.
    ///
    /// Supported paths:
    /// - Update path: PreOperation Update on opportunity when statecode/statuscode changes.
    /// - Won-close path: PostOperation WinOpportunity, because WinOpportunity does not provide an opportunity Target to modify in PreOperation.
    /// - Optional fallback: PostOperation SetState / SetStateDynamicEntity if the org uses an older/custom close path.
    ///
    /// Register Step A:
    /// - Table: opportunity
    /// - Message: Update
    /// - Stage: PreOperation
    /// - Mode: Synchronous
    /// - Filtering Attributes: statecode,statuscode
    /// - PreImage Name: PreImage
    /// - PreImage Fields: statecode,statuscode
    ///
    /// Register Step B:
    /// - Table: opportunity
    /// - Message: WinOpportunity
    /// - Stage: PostOperation
    /// - Mode: Synchronous
    /// - Filtering Attributes: none
    /// - Images: none
    ///
    /// Optional Step C only if your org uses SetState instead of WinOpportunity:
    /// - Table: opportunity
    /// - Message: SetState or SetStateDynamicEntity
    /// - Stage: PostOperation
    /// - Mode: Synchronous
    /// - Images: none
    /// </summary>
    public class OpportunityWonStandardizationPlugin : PluginBase
    {
        protected override void ExecuteInternal(
            IPluginExecutionContext context,
            IOrganizationService service,
            ITracingService trace)
        {
            trace?.Trace("TRACE-PLUGIN3: Starting OpportunityWonStandardizationPlugin.");

            if (context.Depth > 1)
            {
                trace?.Trace("TRACE-PLUGIN3-SKIP: Depth {0} > 1. Exiting to avoid recursion.", context.Depth);
                return;
            }

            if (IsMessage(context, "Update"))
            {
                HandleUpdatePath(context, trace);
                return;
            }

            if (IsMessage(context, "WinOpportunity"))
            {
                HandleWinOpportunityPath(context, service, trace);
                return;
            }

            if (IsMessage(context, "SetState") || IsMessage(context, "SetStateDynamicEntity"))
            {
                HandleSetStatePath(context, service, trace);
                return;
            }

            trace?.Trace(
                "TRACE-PLUGIN3-SKIP: Message {0} is not handled.",
                context.MessageName);
        }

        private static void HandleUpdatePath(
            IPluginExecutionContext context,
            ITracingService trace)
        {
            trace?.Trace("TRACE-PLUGIN3: Handling Update path.");

            if (!IsPreOperation(context) ||
                !IsPrimaryEntity(context, Schema.Opportunity))
            {
                trace?.Trace(
                    "TRACE-PLUGIN3-SKIP: Update path expected Entity=opportunity, Stage=20. Actual Entity={0}, Stage={1}.",
                    context.PrimaryEntityName,
                    context.Stage);
                return;
            }

            var target = GetTargetEntity(context);
            TraceTarget(trace, target);

            if (target == null)
            {
                trace?.Trace("TRACE-PLUGIN3-SKIP: Target was not an Entity.");
                return;
            }

            var preImage = GetPreImage(context);
            TracePreImage(trace, preImage);

            var targetState = target.GetAttributeValue<OptionSetValue>(Schema.OpportunityStateCode);
            var preImageState = preImage?.GetAttributeValue<OptionSetValue>(Schema.OpportunityStateCode);
            var resolvedState = GetAttributeValueFromTargetOrPreImage<OptionSetValue>(target, preImage, Schema.OpportunityStateCode);
            var targetStatus = target.GetAttributeValue<OptionSetValue>(Schema.OpportunityStatusCode);
            var preImageStatus = preImage?.GetAttributeValue<OptionSetValue>(Schema.OpportunityStatusCode);

            trace?.Trace(
                "TRACE-PLUGIN3: Update state/status resolved. TargetState={0}; PreImageState={1}; ResolvedState={2}; TargetStatus={3}; PreImageStatus={4}.",
                FormatOptionSet(targetState),
                FormatOptionSet(preImageState),
                FormatOptionSet(resolvedState),
                FormatOptionSet(targetStatus),
                FormatOptionSet(preImageStatus));

            if (resolvedState == null || resolvedState.Value != Schema.OpportunityStateWon)
            {
                trace?.Trace(
                    "TRACE-PLUGIN3-NOOP: Opportunity is not Won. Expected statecode={0}; Resolved statecode={1}.",
                    Schema.OpportunityStateWon,
                    FormatOptionSet(resolvedState));
                return;
            }

            SetWonProbabilityFields(target);

            trace?.Trace(
                "TRACE-PLUGIN3-ACTION: Update path set {0}=100 and {1}={2}.",
                Schema.OpportunityProbability,
                Schema.OpportunityCustomCloseProbability,
                Schema.CustomCloseProbabilityClosed);
        }

        private static void HandleWinOpportunityPath(
            IPluginExecutionContext context,
            IOrganizationService service,
            ITracingService trace)
        {
            trace?.Trace("TRACE-PLUGIN3: Handling WinOpportunity path.");

            if (!IsPostOperation(context))
            {
                trace?.Trace(
                    "TRACE-PLUGIN3-SKIP: WinOpportunity path must be registered PostOperation stage 40. Actual Stage={0}.",
                    context.Stage);
                return;
            }

            Entity opportunityClose = null;

            if (context.InputParameters != null &&
                context.InputParameters.Contains(Schema.OpportunityCloseInputParameter) &&
                context.InputParameters[Schema.OpportunityCloseInputParameter] is Entity closeEntity)
            {
                opportunityClose = closeEntity;
            }

            if (opportunityClose == null)
            {
                trace?.Trace("TRACE-PLUGIN3-SKIP: WinOpportunity did not include OpportunityClose Entity.");
                return;
            }

            trace?.Trace(
                "TRACE-PLUGIN3: OpportunityClose found. LogicalName={0}; Attributes=[{1}].",
                opportunityClose.LogicalName,
                FormatKeys(opportunityClose.Attributes));

            var opportunityRef = opportunityClose.GetAttributeValue<EntityReference>(Schema.OpportunityCloseOpportunity);

            trace?.Trace(
                "TRACE-PLUGIN3: WinOpportunity Opportunity reference={0}.",
                FormatEntityReference(opportunityRef));

            if (opportunityRef == null || opportunityRef.Id == Guid.Empty)
            {
                trace?.Trace("TRACE-PLUGIN3-SKIP: OpportunityClose did not contain a valid opportunityid.");
                return;
            }

            var update = new Entity(Schema.Opportunity, opportunityRef.Id);
            SetWonProbabilityFields(update);

            service.Update(update);

            trace?.Trace(
                "TRACE-PLUGIN3-ACTION: WinOpportunity path updated opportunity {0}. Set {1}=100 and {2}={3}.",
                opportunityRef.Id,
                Schema.OpportunityProbability,
                Schema.OpportunityCustomCloseProbability,
                Schema.CustomCloseProbabilityClosed);
        }

        private static void HandleSetStatePath(
            IPluginExecutionContext context,
            IOrganizationService service,
            ITracingService trace)
        {
            trace?.Trace("TRACE-PLUGIN3: Handling SetState/SetStateDynamicEntity path.");

            if (!IsPostOperation(context))
            {
                trace?.Trace(
                    "TRACE-PLUGIN3-SKIP: SetState path must be registered PostOperation stage 40. Actual Stage={0}.",
                    context.Stage);
                return;
            }

            EntityReference entityMoniker = null;
            OptionSetValue state = null;

            if (context.InputParameters != null &&
                context.InputParameters.Contains(Schema.EntityMonikerInputParameter) &&
                context.InputParameters[Schema.EntityMonikerInputParameter] is EntityReference moniker)
            {
                entityMoniker = moniker;
            }

            if (context.InputParameters != null &&
                context.InputParameters.Contains(Schema.StateInputParameter) &&
                context.InputParameters[Schema.StateInputParameter] is OptionSetValue stateValue)
            {
                state = stateValue;
            }

            trace?.Trace(
                "TRACE-PLUGIN3: SetState inputs. EntityMoniker={0}; State={1}.",
                FormatEntityReference(entityMoniker),
                FormatOptionSet(state));

            if (entityMoniker == null ||
                entityMoniker.Id == Guid.Empty ||
                !string.Equals(entityMoniker.LogicalName, Schema.Opportunity, StringComparison.OrdinalIgnoreCase))
            {
                trace?.Trace("TRACE-PLUGIN3-SKIP: EntityMoniker is not a valid opportunity reference.");
                return;
            }

            if (state == null || state.Value != Schema.OpportunityStateWon)
            {
                trace?.Trace(
                    "TRACE-PLUGIN3-NOOP: SetState state is not Won. Expected={0}; Actual={1}.",
                    Schema.OpportunityStateWon,
                    FormatOptionSet(state));
                return;
            }

            var update = new Entity(Schema.Opportunity, entityMoniker.Id);
            SetWonProbabilityFields(update);

            service.Update(update);

            trace?.Trace(
                "TRACE-PLUGIN3-ACTION: SetState path updated opportunity {0}. Set {1}=100 and {2}={3}.",
                entityMoniker.Id,
                Schema.OpportunityProbability,
                Schema.OpportunityCustomCloseProbability,
                Schema.CustomCloseProbabilityClosed);
        }
    }
