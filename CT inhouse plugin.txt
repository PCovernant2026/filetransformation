using Microsoft.Xrm.Sdk;
using Microsoft.Xrm.Sdk.Query;
using System;

namespace Ctp.Crm.OpportunityAutomation
{
    /// <summary>
    /// ============================================================
    ///  SCHEMA CONSTANTS
    /// ============================================================
    /// </summary>
    public static class Schema
    {
        // Tables
        public const string Opportunity = "opportunity";
        public const string SystemUser = "systemuser";
        public const string SubSolution = "ctp_subsolutions";

        // Opportunity fields
        public const string OpportunityClientPartner = "new_clientpartnerid";   // Lookup to systemuser
        public const string OpportunityArchitect = "new_architectid";           // Lookup to systemuser
        public const string OpportunitySubSolution = "ctp_subsolutions";        // Lookup to ctp_subsolutions
        public const string OpportunityStage = "new_percentcomplete";           // Stage field on opportunity
        public const string OpportunityStateCode = "statecode";
        public const string OpportunityStatusCode = "statuscode";

        // System User fields
        public const string SystemUserIsClientPartner = "new_clientpartner";    // Boolean / Two Options field

        // Sub-Solution fields
        public const string SubSolutionArchitect = "ctp_architect";             // Lookup to systemuser

        // Opportunity close / state parameters
        public const string OpportunityCloseInputParameter = "OpportunityClose";
        public const string OpportunityCloseOpportunity = "opportunityid";
        public const string EntityMonikerInputParameter = "EntityMoniker";
        public const string StateInputParameter = "State";

        // Dataverse pipeline stages
        public const int PreOperationStage = 20;
        public const int PostOperationStage = 40;

        // Opportunity state values
        public const int OpportunityStateOpen = 0;
        public const int OpportunityStateWon = 1;
        public const int OpportunityStateLost = 2;

        // Stage option value for 100%
        // Target metadata label: 100 - SOW signed
        public const int OpportunityStage100Percent = 100000000;
    }

    /// <summary>
    /// ============================================================
    ///  SHARED PLUGIN BASE
    /// ============================================================
    /// </summary>
    public abstract class PluginBase : IPlugin
    {
        public void Execute(IServiceProvider serviceProvider)
        {
            if (serviceProvider == null)
            {
                throw new InvalidPluginExecutionException("Service provider is null.");
            }

            var context = (IPluginExecutionContext)serviceProvider.GetService(typeof(IPluginExecutionContext));
            var trace = (ITracingService)serviceProvider.GetService(typeof(ITracingService));
            var serviceFactory = (IOrganizationServiceFactory)serviceProvider.GetService(typeof(IOrganizationServiceFactory));

            if (context == null)
            {
                throw new InvalidPluginExecutionException("Plugin execution context is null.");
            }

            if (serviceFactory == null)
            {
                throw new InvalidPluginExecutionException("Organization service factory is null.");
            }

            // System context for reads and PostOperation update.
            var service = serviceFactory.CreateOrganizationService(null);

            try
            {
                TraceExecutionStart(context, trace, GetType().FullName);

                ExecuteInternal(context, service, trace);

                trace?.Trace(
                    "TRACE-END: {0} completed successfully. CorrelationId={1}",
                    GetType().FullName,
                    context.CorrelationId);
            }
            catch (InvalidPluginExecutionException ex)
            {
                trace?.Trace(
                    "TRACE-ERROR: InvalidPluginExecutionException in {0}. Message={1}. CorrelationId={2}",
                    GetType().FullName,
                    ex.Message,
                    context.CorrelationId);

                throw;
            }
            catch (Exception ex)
            {
                trace?.Trace(
                    "TRACE-ERROR: Unhandled exception in {0}. CorrelationId={1}. Exception={2}",
                    GetType().FullName,
                    context.CorrelationId,
                    ex);

                throw new InvalidPluginExecutionException(
                    $"An error occurred in {GetType().Name}: {ex.Message}",
                    ex);
            }
        }

        protected abstract void ExecuteInternal(
            IPluginExecutionContext context,
            IOrganizationService service,
            ITracingService trace);

        protected static Entity GetTargetEntity(IPluginExecutionContext context)
        {
            if (context.InputParameters != null &&
                context.InputParameters.Contains("Target") &&
                context.InputParameters["Target"] is Entity target)
            {
                return target;
            }

            return null;
        }

        protected static Entity GetPreImage(IPluginExecutionContext context, string imageName = "PreImage")
        {
            if (context.PreEntityImages != null &&
                context.PreEntityImages.Contains(imageName))
            {
                return context.PreEntityImages[imageName];
            }

            return null;
        }

        protected static T GetAttributeValueFromTargetOrPreImage<T>(
            Entity target,
            Entity preImage,
            string attributeName)
        {
            if (target != null && target.Attributes.Contains(attributeName))
            {
                return target.GetAttributeValue<T>(attributeName);
            }

            if (preImage != null && preImage.Attributes.Contains(attributeName))
            {
                return preImage.GetAttributeValue<T>(attributeName);
            }

            return default(T);
        }

        protected static bool IsMessage(IPluginExecutionContext context, string messageName)
        {
            return string.Equals(context.MessageName, messageName, StringComparison.OrdinalIgnoreCase);
        }

        protected static bool IsPrimaryEntity(IPluginExecutionContext context, string entityName)
        {
            return string.Equals(context.PrimaryEntityName, entityName, StringComparison.OrdinalIgnoreCase);
        }

        protected static bool IsPreOperation(IPluginExecutionContext context)
        {
            return context.Stage == Schema.PreOperationStage;
        }

        protected static bool IsPostOperation(IPluginExecutionContext context)
        {
            return context.Stage == Schema.PostOperationStage;
        }

        protected static void SetOpportunityStageTo100Percent(Entity opportunity)
        {
            if (opportunity == null)
            {
                return;
            }

            opportunity[Schema.OpportunityStage] =
                new OptionSetValue(Schema.OpportunityStage100Percent);
        }

        protected static void TraceExecutionStart(
            IPluginExecutionContext context,
            ITracingService trace,
            string pluginName)
        {
            trace?.Trace("TRACE-START: {0}", pluginName);

            trace?.Trace(
                "TRACE-CONTEXT: Message={0}; PrimaryEntity={1}; PrimaryEntityId={2}; Stage={3}; Mode={4}; Depth={5}; UserId={6}; InitiatingUserId={7}; OrganizationId={8}; CorrelationId={9}; OperationId={10}",
                context.MessageName,
                context.PrimaryEntityName,
                context.PrimaryEntityId,
                context.Stage,
                context.Mode,
                context.Depth,
                context.UserId,
                context.InitiatingUserId,
                context.OrganizationId,
                context.CorrelationId,
                context.OperationId);

            trace?.Trace(
                "TRACE-PARAMETERS: InputParameters=[{0}]; OutputParameters=[{1}]; SharedVariables=[{2}]",
                FormatKeys(context.InputParameters),
                FormatKeys(context.OutputParameters),
                FormatKeys(context.SharedVariables));
        }

        protected static void TraceTarget(ITracingService trace, Entity target)
        {
            if (target == null)
            {
                trace?.Trace("TRACE-TARGET: Target is null.");
                return;
            }

            trace?.Trace(
                "TRACE-TARGET: LogicalName={0}; Id={1}; AttributeCount={2}; Attributes=[{3}]",
                target.LogicalName,
                target.Id,
                target.Attributes.Count,
                FormatKeys(target.Attributes));
        }

        protected static void TracePreImage(
            ITracingService trace,
            Entity preImage,
            string imageName = "PreImage")
        {
            if (preImage == null)
            {
                trace?.Trace("TRACE-PREIMAGE: {0} was not available.", imageName);
                return;
            }

            trace?.Trace(
                "TRACE-PREIMAGE: {0}; LogicalName={1}; Id={2}; AttributeCount={3}; Attributes=[{4}]",
                imageName,
                preImage.LogicalName,
                preImage.Id,
                preImage.Attributes.Count,
                FormatKeys(preImage.Attributes));
        }

        // This single method prevents AttributeCollection ambiguity.
        protected static string FormatKeys(Microsoft.Xrm.Sdk.DataCollection<string, object> collection)
        {
            if (collection == null || collection.Count == 0)
            {
                return "none";
            }

            return string.Join(", ", collection.Keys);
        }

        protected static string FormatEntityReference(EntityReference entityReference)
        {
            if (entityReference == null)
            {
                return "null";
            }

            return $"{entityReference.LogicalName}:{entityReference.Id}";
        }

        protected static string FormatOptionSet(OptionSetValue optionSetValue)
        {
            return optionSetValue == null ? "null" : optionSetValue.Value.ToString();
        }
    }

    /// <summary>
    /// ============================================================
    ///  PLUGIN 1:
    ///  OpportunityClientPartnerFromUserFlagPlugin
    ///
    ///  Business rule:
    ///  - On Opportunity Create
    ///  - Query creating user
    ///  - If systemuser.new_clientpartner = true
    ///  - Populate opportunity.new_clientpartnerid
    ///
    ///  Register:
    ///  - Table: opportunity
    ///  - Message: Create
    ///  - Stage: PreOperation
    ///  - Mode: Synchronous
    /// ============================================================
    /// </summary>
    public class OpportunityClientPartnerFromUserFlagPlugin : PluginBase
    {
        protected override void ExecuteInternal(
            IPluginExecutionContext context,
            IOrganizationService service,
            ITracingService trace)
        {
            trace?.Trace("TRACE-PARTNER: Starting OpportunityClientPartnerFromUserFlagPlugin.");

            if (context.Depth > 1)
            {
                trace?.Trace("TRACE-PARTNER-SKIP: Depth {0} > 1. Exiting.", context.Depth);
                return;
            }

            if (!IsPreOperation(context) ||
                !IsMessage(context, "Create") ||
                !IsPrimaryEntity(context, Schema.Opportunity))
            {
                trace?.Trace(
                    "TRACE-PARTNER-SKIP: Expected Create opportunity PreOperation. Actual Message={0}; Entity={1}; Stage={2}.",
                    context.MessageName,
                    context.PrimaryEntityName,
                    context.Stage);
                return;
            }

            var target = GetTargetEntity(context);
            TraceTarget(trace, target);

            if (target == null)
            {
                trace?.Trace("TRACE-PARTNER-SKIP: Target is null.");
                return;
            }

            if (target.Attributes.Contains(Schema.OpportunityClientPartner) &&
                target[Schema.OpportunityClientPartner] != null)
            {
                trace?.Trace(
                    "TRACE-PARTNER-NOOP: {0} already supplied as {1}. No overwrite.",
                    Schema.OpportunityClientPartner,
                    FormatEntityReference(target.GetAttributeValue<EntityReference>(Schema.OpportunityClientPartner)));
                return;
            }

            var creatingUserId = context.InitiatingUserId != Guid.Empty
                ? context.InitiatingUserId
                : context.UserId;

            if (creatingUserId == Guid.Empty)
            {
                trace?.Trace("TRACE-PARTNER-SKIP: Creating user could not be resolved.");
                return;
            }

            trace?.Trace(
                "TRACE-PARTNER: Retrieving systemuser {0} with column {1}.",
                creatingUserId,
                Schema.SystemUserIsClientPartner);

            var creatingUser = service.Retrieve(
                Schema.SystemUser,
                creatingUserId,
                new ColumnSet(Schema.SystemUserIsClientPartner));

            if (creatingUser == null)
            {
                trace?.Trace("TRACE-PARTNER-SKIP: Creating user record could not be retrieved.");
                return;
            }

            bool isClientPartner =
                creatingUser.Contains(Schema.SystemUserIsClientPartner) &&
                creatingUser.GetAttributeValue<bool>(Schema.SystemUserIsClientPartner);

            trace?.Trace(
                "TRACE-PARTNER: Creating user {0}; {1}={2}.",
                creatingUserId,
                Schema.SystemUserIsClientPartner,
                isClientPartner);

            if (!isClientPartner)
            {
                trace?.Trace(
                    "TRACE-PARTNER-NOOP: User is not marked as Client Partner. {0} not populated.",
                    Schema.OpportunityClientPartner);
                return;
            }

            var userRef = new EntityReference(Schema.SystemUser, creatingUserId);
            target[Schema.OpportunityClientPartner] = userRef;

            trace?.Trace(
                "TRACE-PARTNER-ACTION: User is Client Partner. Defaulted {0} to {1}.",
                Schema.OpportunityClientPartner,
                FormatEntityReference(userRef));
        }
    }

    /// <summary>
    /// ============================================================
    ///  PLUGIN 2:
    ///  OpportunityArchitectFromSubSolutionPlugin
    ///
    ///  Business rule:
    ///  - On Opportunity Create/Update
    ///  - Use opportunity.ctp_subsolutions lookup
    ///  - Query ctp_subsolutions.ctp_architect
    ///  - Populate opportunity.new_architectid
    ///
    ///  Register Create:
    ///  - Table: opportunity
    ///  - Message: Create
    ///  - Stage: PreOperation
    ///
    ///  Register Update:
    ///  - Table: opportunity
    ///  - Message: Update
    ///  - Stage: PreOperation
    ///  - Filtering Attributes: ctp_subsolutions
    /// ============================================================
    /// </summary>
    public class OpportunityArchitectFromSubSolutionPlugin : PluginBase
    {
        protected override void ExecuteInternal(
            IPluginExecutionContext context,
            IOrganizationService service,
            ITracingService trace)
        {
            trace?.Trace("TRACE-ARCHITECT: Starting OpportunityArchitectFromSubSolutionPlugin.");

            if (context.Depth > 1)
            {
                trace?.Trace("TRACE-ARCHITECT-SKIP: Depth {0} > 1. Exiting.", context.Depth);
                return;
            }

            if (!IsPreOperation(context) ||
                !IsPrimaryEntity(context, Schema.Opportunity))
            {
                trace?.Trace(
                    "TRACE-ARCHITECT-SKIP: Expected opportunity PreOperation. Actual Entity={0}; Stage={1}.",
                    context.PrimaryEntityName,
                    context.Stage);
                return;
            }

            if (!IsMessage(context, "Create") &&
                !IsMessage(context, "Update"))
            {
                trace?.Trace(
                    "TRACE-ARCHITECT-SKIP: Message {0} is not handled.",
                    context.MessageName);
                return;
            }

            var target = GetTargetEntity(context);
            TraceTarget(trace, target);

            if (target == null)
            {
                trace?.Trace("TRACE-ARCHITECT-SKIP: Target is null.");
                return;
            }

            if (IsMessage(context, "Update") &&
                !target.Attributes.Contains(Schema.OpportunitySubSolution))
            {
                trace?.Trace(
                    "TRACE-ARCHITECT-SKIP: Update did not include {0}.",
                    Schema.OpportunitySubSolution);
                return;
            }

            if (target.Attributes.Contains(Schema.OpportunityArchitect))
            {
                trace?.Trace(
                    "TRACE-ARCHITECT-NOOP: {0} explicitly supplied as {1}. Preserving.",
                    Schema.OpportunityArchitect,
                    FormatEntityReference(target.GetAttributeValue<EntityReference>(Schema.OpportunityArchitect)));
                return;
            }

            if (!target.Attributes.Contains(Schema.OpportunitySubSolution))
            {
                trace?.Trace(
                    "TRACE-ARCHITECT-SKIP: Target does not contain {0}. Nothing to map.",
                    Schema.OpportunitySubSolution);
                return;
            }

            var subSolutionRef = target.GetAttributeValue<EntityReference>(
                Schema.OpportunitySubSolution);

            trace?.Trace(
                "TRACE-ARCHITECT: {0}={1}.",
                Schema.OpportunitySubSolution,
                FormatEntityReference(subSolutionRef));

            if (subSolutionRef == null)
            {
                target[Schema.OpportunityArchitect] = null;

                trace?.Trace(
                    "TRACE-ARCHITECT-ACTION: Sub-Solution cleared. Cleared {0}.",
                    Schema.OpportunityArchitect);
                return;
            }

            var subSolution = service.Retrieve(
                Schema.SubSolution,
                subSolutionRef.Id,
                new ColumnSet(Schema.SubSolutionArchitect));

            var architectRef = subSolution?.GetAttributeValue<EntityReference>(
                Schema.SubSolutionArchitect);

            trace?.Trace(
                "TRACE-ARCHITECT: Retrieved {0}.{1}={2}.",
                Schema.SubSolution,
                Schema.SubSolutionArchitect,
                FormatEntityReference(architectRef));

            if (architectRef == null)
            {
                target[Schema.OpportunityArchitect] = null;

                trace?.Trace(
                    "TRACE-ARCHITECT-ACTION: Selected Sub-Solution has no architect. Cleared {0}.",
                    Schema.OpportunityArchitect);
                return;
            }

            target[Schema.OpportunityArchitect] = architectRef;

            trace?.Trace(
                "TRACE-ARCHITECT-ACTION: Defaulted {0} to {1}.",
                Schema.OpportunityArchitect,
                FormatEntityReference(architectRef));
        }
    }

    /// <summary>
    /// ============================================================
    ///  PLUGIN 3:
    ///  OpportunityWonStagePlugin
    ///
    ///  Business rule:
    ///  - When Opportunity is won
    ///  - Set opportunity.new_percentcomplete to 100%
    ///
    ///  Supports:
    ///  - Update
    ///  - Win
    ///  - WinOpportunity
    ///  - SetState / SetStateDynamicEntity
    /// ============================================================
    /// </summary>
    public class OpportunityWonStagePlugin : PluginBase
    {
        protected override void ExecuteInternal(
            IPluginExecutionContext context,
            IOrganizationService service,
            ITracingService trace)
        {
            trace?.Trace("TRACE-WONSTAGE: Starting OpportunityWonStagePlugin.");

            if (context.Depth > 1)
            {
                trace?.Trace("TRACE-WONSTAGE-SKIP: Depth {0} > 1. Exiting.", context.Depth);
                return;
            }

            if (IsMessage(context, "Update"))
            {
                HandleUpdatePath(context, trace);
                return;
            }

            if (IsMessage(context, "Win") || IsMessage(context, "WinOpportunity"))
            {
                HandleWinPath(context, service, trace);
                return;
            }

            if (IsMessage(context, "SetState") ||
                IsMessage(context, "SetStateDynamicEntity"))
            {
                HandleSetStatePath(context, service, trace);
                return;
            }

            trace?.Trace(
                "TRACE-WONSTAGE-SKIP: Message {0} is not handled.",
                context.MessageName);
        }

        private static void HandleUpdatePath(
            IPluginExecutionContext context,
            ITracingService trace)
        {
            trace?.Trace("TRACE-WONSTAGE: Handling Update path.");

            if (!IsPreOperation(context) ||
                !IsPrimaryEntity(context, Schema.Opportunity))
            {
                trace?.Trace(
                    "TRACE-WONSTAGE-SKIP: Update path expected opportunity PreOperation. Actual Entity={0}; Stage={1}.",
                    context.PrimaryEntityName,
                    context.Stage);
                return;
            }

            var target = GetTargetEntity(context);
            TraceTarget(trace, target);

            if (target == null)
            {
                trace?.Trace("TRACE-WONSTAGE-SKIP: Target is null.");
                return;
            }

            var preImage = GetPreImage(context);
            TracePreImage(trace, preImage);

            var targetState = target.GetAttributeValue<OptionSetValue>(
                Schema.OpportunityStateCode);

            var preImageState = preImage?.GetAttributeValue<OptionSetValue>(
                Schema.OpportunityStateCode);

            var resolvedState = GetAttributeValueFromTargetOrPreImage<OptionSetValue>(
                target,
                preImage,
                Schema.OpportunityStateCode);

            trace?.Trace(
                "TRACE-WONSTAGE: TargetState={0}; PreImageState={1}; ResolvedState={2}.",
                FormatOptionSet(targetState),
                FormatOptionSet(preImageState),
                FormatOptionSet(resolvedState));

            if (resolvedState == null ||
                resolvedState.Value != Schema.OpportunityStateWon)
            {
                trace?.Trace(
                    "TRACE-WONSTAGE-NOOP: Opportunity is not Won. Expected={0}; Actual={1}.",
                    Schema.OpportunityStateWon,
                    FormatOptionSet(resolvedState));
                return;
            }

            SetOpportunityStageTo100Percent(target);

            trace?.Trace(
                "TRACE-WONSTAGE-ACTION: Update path set {0}={1}.",
                Schema.OpportunityStage,
                Schema.OpportunityStage100Percent);
        }

        private static void HandleWinPath(
            IPluginExecutionContext context,
            IOrganizationService service,
            ITracingService trace)
        {
            trace?.Trace(
                "TRACE-WONSTAGE: Handling Win path. Message={0}; Stage={1}.",
                context.MessageName,
                context.Stage);

            Entity opportunityClose = null;

            if (context.InputParameters != null &&
                context.InputParameters.Contains(Schema.OpportunityCloseInputParameter) &&
                context.InputParameters[Schema.OpportunityCloseInputParameter] is Entity closeEntity)
            {
                opportunityClose = closeEntity;
            }

            if (opportunityClose == null)
            {
                trace?.Trace("TRACE-WONSTAGE-SKIP: OpportunityClose input was not found.");
                return;
            }

            trace?.Trace(
                "TRACE-WONSTAGE: OpportunityClose found. LogicalName={0}; Attributes=[{1}].",
                opportunityClose.LogicalName,
                FormatKeys(opportunityClose.Attributes));

            var opportunityRef = opportunityClose.GetAttributeValue<EntityReference>(
                Schema.OpportunityCloseOpportunity);

            trace?.Trace(
                "TRACE-WONSTAGE: Opportunity reference={0}.",
                FormatEntityReference(opportunityRef));

            if (opportunityRef == null ||
                opportunityRef.Id == Guid.Empty)
            {
                trace?.Trace("TRACE-WONSTAGE-SKIP: Invalid opportunity reference.");
                return;
            }

            var update = new Entity(Schema.Opportunity, opportunityRef.Id);
            SetOpportunityStageTo100Percent(update);

            service.Update(update);

            trace?.Trace(
                "TRACE-WONSTAGE-ACTION: Win path updated opportunity {0}. Set {1}={2}.",
                opportunityRef.Id,
                Schema.OpportunityStage,
                Schema.OpportunityStage100Percent);
        }

        private static void HandleSetStatePath(
            IPluginExecutionContext context,
            IOrganizationService service,
            ITracingService trace)
        {
            trace?.Trace("TRACE-WONSTAGE: Handling SetState / SetStateDynamicEntity path.");

            if (!IsPostOperation(context))
            {
                trace?.Trace(
                    "TRACE-WONSTAGE-SKIP: SetState must be PostOperation stage 40. Actual Stage={0}.",
                    context.Stage);
                return;
            }

            EntityReference entityMoniker = null;
            OptionSetValue state = null;

            if (context.InputParameters != null &&
                context.InputParameters.Contains(Schema.EntityMonikerInputParameter) &&
                context.InputParameters[Schema.EntityMonikerInputParameter] is EntityReference moniker)
            {
                entityMoniker = moniker;
            }

            if (context.InputParameters != null &&
                context.InputParameters.Contains(Schema.StateInputParameter) &&
                context.InputParameters[Schema.StateInputParameter] is OptionSetValue stateValue)
            {
                state = stateValue;
            }

            trace?.Trace(
                "TRACE-WONSTAGE: EntityMoniker={0}; State={1}.",
                FormatEntityReference(entityMoniker),
                FormatOptionSet(state));

            if (entityMoniker == null ||
                entityMoniker.Id == Guid.Empty ||
                !string.Equals(entityMoniker.LogicalName, Schema.Opportunity, StringComparison.OrdinalIgnoreCase))
            {
                trace?.Trace("TRACE-WONSTAGE-SKIP: EntityMoniker is not a valid opportunity.");
                return;
            }

            if (state == null ||
                state.Value != Schema.OpportunityStateWon)
            {
                trace?.Trace(
                    "TRACE-WONSTAGE-NOOP: State is not Won. Expected={0}; Actual={1}.",
                    Schema.OpportunityStateWon,
                    FormatOptionSet(state));
                return;
            }

            var update = new Entity(Schema.Opportunity, entityMoniker.Id);
            SetOpportunityStageTo100Percent(update);

            service.Update(update);

            trace?.Trace(
                "TRACE-WONSTAGE-ACTION: SetState updated opportunity {0}. Set {1}={2}.",
                entityMoniker.Id,
                Schema.OpportunityStage,
                Schema.OpportunityStage100Percent);
        }
    }
