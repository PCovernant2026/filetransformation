namespace CTP.VendorQuoteAgent

    /// <summary>
    /// Dataverse Custom API plug-in for ctp_InvokeVendorQuoteAgent.
    ///
    /// This v6 version improves the Direct Line bridge by:
    /// - Sending the exact API marker + JSON structure expected by the Copilot Agent.
    /// - Draining initial welcome/greeting activities before sending the request.
    /// - Ignoring non-JSON bot messages instead of failing immediately.
    /// - Extracting the first valid JSON object if the agent wraps JSON in text/code fences.
    /// - Returning a controlled JSON error with rawAgentResponse if no JSON is found.
    /// - Writing detailed Plug-in Trace Logs for the portal request, exact agent message, Direct Line activity, and bot responses.
    /// - Supporting operation=debug_echo_agent_message to return the exact agent message without calling Direct Line.
    /// - Normalizing double-encoded JSON strings before sending to the agent.
    /// - Returning a clear error if the Custom API body is not wrapped in ctp_RequestJson.
    /// - Supporting operation=directline_diagnostic to call Direct Line and return every activity received.
    /// - Including all captured non-JSON bot messages in Invalid Agent Response diagnostics.
    ///
    /// Exact message sent to agent:
    /// CTP_VENDOR_QUOTE_AGENT_REQUEST
    /// {ctp_RequestJson}
    ///
    /// Required Custom API input:  ctp_RequestJson  (String)
    /// Required Custom API output: ctp_ResponseJson (String)
    ///
    /// Required environment variables:
    /// - ctp_CopilotDirectLineSecret        Text for dev; secure later for production
    /// - ctp_CopilotDirectLineBaseUrl       Text, usually https://directline.botframework.com
    /// - ctp_CopilotBridgeTimeoutSeconds    Decimal/Text, recommended 25
    /// - ctp_CopilotAgentName               Text, optional, used for tracing only
    /// </summary>
    public class InvokeVendorQuoteAgentPlugin : IPlugin
    {
        private const string InputRequestJson = "ctp_RequestJson";
        private const string OutputResponseJson = "ctp_ResponseJson";
        private const string PluginSenderId = "dataverse-custom-api-plugin";
        private const int RawAgentResponseMaxLength = 1500;
        private const int TraceChunkSize = 3000;
        private const int DiagnosticActivityTextMaxLength = 5000;
        private const int DiagnosticMaxActivities = 50;

        public void Execute(IServiceProvider serviceProvider)
        {
            var context = (IPluginExecutionContext)serviceProvider.GetService(typeof(IPluginExecutionContext));
            var tracing = (ITracingService)serviceProvider.GetService(typeof(ITracingService));
            var serviceFactory = (IOrganizationServiceFactory)serviceProvider.GetService(typeof(IOrganizationServiceFactory));
            var service = serviceFactory.CreateOrganizationService(context.UserId);

            try
            {
                tracing.Trace("ctp_InvokeVendorQuoteAgent plug-in started.");

                TraceInputParameterKeys(context, tracing);

                string rawRequestJson = GetRequestJsonFromContext(context, tracing);
                if (string.IsNullOrWhiteSpace(rawRequestJson))
                {
                    context.OutputParameters[OutputResponseJson] = BuildApiErrorJson(
                        string.Empty,
                        "Missing Request",
                        "Missing ctp_RequestJson. The Dataverse Custom API body must be { \"ctp_RequestJson\": \"{...}\" }.");
                    return;
                }

                string requestJson = NormalizeRequestJson(rawRequestJson, tracing);
                if (string.IsNullOrWhiteSpace(requestJson) || !LooksLikeJsonObject(requestJson) || !IsValidJson(requestJson))
                {
                    context.OutputParameters[OutputResponseJson] = BuildApiErrorJson(
                        string.Empty,
                        "Invalid Request",
                        "ctp_RequestJson was received, but it was not a valid JSON object after normalization.");
                    return;
                }

                string operation = TryExtractOperation(requestJson);

                tracing.Trace("Context MessageName: " + context.MessageName);
                tracing.Trace("Context CorrelationId: " + context.CorrelationId.ToString());
                tracing.Trace("Operation: " + operation);
                tracing.Trace("Raw request length: " + rawRequestJson.Length.ToString());
                tracing.Trace("Normalized request length: " + requestJson.Length.ToString());
                TraceLarge(tracing, "Raw request from Embedded Portal", rawRequestJson);
                TraceLarge(tracing, "Normalized ctp_RequestJson used for agent", requestJson);

                // Temporary debug operation: returns the exact message that would be sent to the Copilot Agent
                // without calling Direct Line. Useful to verify the API marker + JSON format from the portal.
                if (string.Equals(operation, "debug_echo_agent_message", StringComparison.OrdinalIgnoreCase))
                {
                    string debugAgentMessage = BuildStrictApiModeMessage(requestJson);
                    tracing.Trace("debug_echo_agent_message requested. Returning agent message without calling Direct Line.");
                    TraceLarge(tracing, "Exact agent message that would be sent", debugAgentMessage);

                    context.OutputParameters[OutputResponseJson] = "{"
                        + "\"success\":true,"
                        + "\"operation\":\"debug_echo_agent_message\","
                        + "\"status\":\"Debug Echo\","
                        + "\"message\":\"This is the exact text the plug-in would send to the Copilot Agent.\","
                        + "\"requestJson\":" + Quote(requestJson) + ","
                        + "\"agentMessage\":" + Quote(debugAgentMessage)
                        + "}";
                    return;
                }

                string directLineSecret = GetEnvironmentVariable(service, "ctp_CopilotDirectLineSecret");
                string directLineBaseUrl = GetEnvironmentVariable(service, "ctp_CopilotDirectLineBaseUrl", required: false);
                string timeoutValue = GetEnvironmentVariable(service, "ctp_CopilotBridgeTimeoutSeconds", required: false);
                string agentName = GetEnvironmentVariable(service, "ctp_CopilotAgentName", required: false);

                if (string.IsNullOrWhiteSpace(directLineSecret))
                {
                    context.OutputParameters[OutputResponseJson] = BuildApiErrorJson(operation, "Missing Configuration", "Missing environment variable: ctp_CopilotDirectLineSecret.");
                    return;
                }

                if (string.IsNullOrWhiteSpace(directLineBaseUrl))
                {
                    directLineBaseUrl = "https://directline.botframework.com";
                }

                int timeoutSeconds = 25;
                int parsedTimeout;
                if (int.TryParse(timeoutValue, out parsedTimeout) && parsedTimeout > 0)
                {
                    timeoutSeconds = parsedTimeout;
                }

                tracing.Trace("Agent Name: " + (agentName ?? string.Empty));
                tracing.Trace("Direct Line Base URL: " + directLineBaseUrl);
                tracing.Trace("Timeout Seconds: " + timeoutSeconds.ToString());
                tracing.Trace("Operation: " + operation);

                if (string.Equals(operation, "directline_diagnostic", StringComparison.OrdinalIgnoreCase))
                {
                    tracing.Trace("directline_diagnostic requested. Calling Direct Line and capturing all returned activities.");
                    string diagnosticJson = RunDirectLineDiagnostic(
                        directLineBaseUrl.TrimEnd('/'),
                        directLineSecret,
                        requestJson,
                        timeoutSeconds,
                        agentName,
                        tracing);

                    context.OutputParameters[OutputResponseJson] = diagnosticJson;
                    tracing.Trace("ctp_InvokeVendorQuoteAgent plug-in completed with directline_diagnostic result.");
                    return;
                }

                string agentResponseJson = SendRequestToCopilotAgent(
                    directLineBaseUrl.TrimEnd('/'),
                    directLineSecret,
                    requestJson,
                    operation,
                    timeoutSeconds,
                    tracing);

                context.OutputParameters[OutputResponseJson] = agentResponseJson;
                tracing.Trace("ctp_InvokeVendorQuoteAgent plug-in completed.");
            }
            catch (Exception ex)
            {
                tracing.Trace("Unhandled error: " + ex.ToString());
                context.OutputParameters[OutputResponseJson] = BuildApiErrorJson("", "Plug-in Error", "Plug-in error: " + ex.Message);
            }
        }

        private static string RunDirectLineDiagnostic(string baseUrl, string secret, string requestJson, int timeoutSeconds, string agentName, ITracingService tracing)
        {
            string probeRequestJson = BuildDiagnosticProbeRequestJson(requestJson);
            string outgoingMessage = BuildStrictApiModeMessage(probeRequestJson);

            var initialActivities = new List<Activity>();
            var responseActivities = new List<Activity>();

            bool conversationStarted = false;
            bool messageSent = false;
            string conversationId = string.Empty;
            string watermark = null;

            try
            {
                var conversation = StartConversation(baseUrl, secret, timeoutSeconds, tracing);
                if (conversation == null || string.IsNullOrWhiteSpace(conversation.ConversationId))
                {
                    return BuildApiErrorJson("directline_diagnostic", "Direct Line Error", "Failed to start Direct Line conversation.");
                }

                conversationStarted = true;
                conversationId = conversation.ConversationId;

                try
                {
                    Thread.Sleep(700);
                    var initial = GetActivities(baseUrl, secret, conversationId, null, timeoutSeconds, tracing);
                    if (initial != null)
                    {
                        watermark = initial.Watermark;
                        if (initial.Activities != null)
                        {
                            AddActivities(initialActivities, initial.Activities);
                        }
                        tracing.Trace("Diagnostic initial activities drained. Watermark: " + (watermark ?? string.Empty));
                        TraceActivities(tracing, "Diagnostic initial activity", initial.Activities);
                    }
                }
                catch (Exception drainEx)
                {
                    tracing.Trace("Diagnostic initial drain warning: " + drainEx.Message);
                }

                var activity = new OutboundActivity
                {
                    Type = "message",
                    From = new ChannelAccount
                    {
                        Id = PluginSenderId,
                        Name = "Dynamics 365 Embedded Portal"
                    },
                    Text = outgoingMessage
                };

                TraceLarge(tracing, "Diagnostic exact outgoing agent message", outgoingMessage);
                SendActivity(baseUrl, secret, conversationId, activity, timeoutSeconds, tracing);
                messageSent = true;

                DateTime stopAt = DateTime.UtcNow.AddSeconds(timeoutSeconds);
                DateTime? lastNewNonPluginActivityAt = null;
                int quietSecondsAfterFirstResponse = 4;

                while (DateTime.UtcNow < stopAt)
                {
                    Thread.Sleep(1000);

                    var polled = GetActivities(baseUrl, secret, conversationId, watermark, timeoutSeconds, tracing);
                    if (polled == null)
                    {
                        continue;
                    }

                    watermark = polled.Watermark;
                    if (polled.Activities == null || polled.Activities.Count == 0)
                    {
                        if (lastNewNonPluginActivityAt.HasValue && DateTime.UtcNow >= lastNewNonPluginActivityAt.Value.AddSeconds(quietSecondsAfterFirstResponse))
                        {
                            break;
                        }
                        continue;
                    }

                    TraceActivities(tracing, "Diagnostic polled activity", polled.Activities);

                    bool sawNewNonPluginActivity = false;
                    for (int i = 0; i < polled.Activities.Count; i++)
                    {
                        var a = polled.Activities[i];
                        if (a == null) continue;

                        if (responseActivities.Count < DiagnosticMaxActivities)
                        {
                            responseActivities.Add(a);
                        }

                        bool isFromPlugin = a.From != null && string.Equals(a.From.Id, PluginSenderId, StringComparison.OrdinalIgnoreCase);
                        if (!isFromPlugin)
                        {
                            sawNewNonPluginActivity = true;
                        }
                    }

                    if (sawNewNonPluginActivity)
                    {
                        lastNewNonPluginActivityAt = DateTime.UtcNow;
                    }

                    if (lastNewNonPluginActivityAt.HasValue && DateTime.UtcNow >= lastNewNonPluginActivityAt.Value.AddSeconds(quietSecondsAfterFirstResponse))
                    {
                        break;
                    }
                }

                return BuildDirectLineDiagnosticJson(
                    agentName,
                    conversationStarted,
                    conversationId,
                    messageSent,
                    probeRequestJson,
                    outgoingMessage,
                    initialActivities,
                    responseActivities);
            }
            catch (Exception ex)
            {
                tracing.Trace("directline_diagnostic error: " + ex.ToString());
                return "{"
                    + "\"success\":false,"
                    + "\"operation\":\"directline_diagnostic\","
                    + "\"status\":\"Diagnostic Error\","
                    + "\"message\":" + Quote(ex.Message) + ","
                    + "\"configuredAgentName\":" + Quote(agentName ?? string.Empty) + ","
                    + "\"conversationStarted\":" + (conversationStarted ? "true" : "false") + ","
                    + "\"conversationId\":" + Quote(conversationId ?? string.Empty) + ","
                    + "\"messageSent\":" + (messageSent ? "true" : "false") + ","
                    + "\"payload\":{\"header\":{},\"lines\":[]}"
                    + "}";
            }
        }

        private static string BuildDiagnosticProbeRequestJson(string requestJson)
        {
            if (string.IsNullOrWhiteSpace(requestJson))
            {
                return "{\"operation\":\"test_connection\",\"source\":\"Dataverse Plug-in Direct Line Diagnostic\"}";
            }

            string replaced = Regex.Replace(
                requestJson,
                "\"operation\"\\s*:\\s*\"directline_diagnostic\"",
                "\"operation\":\"test_connection\"",
                RegexOptions.IgnoreCase);

            return replaced;
        }

        private static void AddActivities(List<Activity> destination, List<Activity> source)
        {
            if (destination == null || source == null) return;

            for (int i = 0; i < source.Count && destination.Count < DiagnosticMaxActivities; i++)
            {
                if (source[i] != null)
                {
                    destination.Add(source[i]);
                }
            }
        }

        private static string BuildDirectLineDiagnosticJson(
            string agentName,
            bool conversationStarted,
            string conversationId,
            bool messageSent,
            string probeRequestJson,
            string outgoingMessage,
            List<Activity> initialActivities,
            List<Activity> responseActivities)
        {
            int validJsonCount = 0;
            int nonJsonBotMessageCount = 0;
            string firstValidJson = string.Empty;

            if (responseActivities != null)
            {
                for (int i = 0; i < responseActivities.Count; i++)
                {
                    var a = responseActivities[i];
                    if (a == null) continue;
                    if (!string.Equals(a.Type, "message", StringComparison.OrdinalIgnoreCase)) continue;
                    if (string.IsNullOrWhiteSpace(a.Text)) continue;

                    bool isFromPlugin = a.From != null && string.Equals(a.From.Id, PluginSenderId, StringComparison.OrdinalIgnoreCase);
                    if (isFromPlugin) continue;

                    string extracted = TryExtractValidJson(a.Text);
                    if (!string.IsNullOrWhiteSpace(extracted))
                    {
                        validJsonCount++;
                        if (string.IsNullOrWhiteSpace(firstValidJson))
                        {
                            firstValidJson = extracted;
                        }
                    }
                    else
                    {
                        nonJsonBotMessageCount++;
                    }
                }
            }

            var sb = new StringBuilder();
            sb.Append("{");
            sb.Append("\"success\":true,");
            sb.Append("\"operation\":\"directline_diagnostic\",");
            sb.Append("\"status\":\"Direct Line Diagnostic\",");
            sb.Append("\"message\":\"Direct Line was called and every captured activity is included below. No Direct Line secret is returned.\",");
            sb.Append("\"configuredAgentName\":").Append(Quote(agentName ?? string.Empty)).Append(",");
            sb.Append("\"conversationStarted\":").Append(conversationStarted ? "true" : "false").Append(",");
            sb.Append("\"conversationId\":").Append(Quote(conversationId ?? string.Empty)).Append(",");
            sb.Append("\"messageSent\":").Append(messageSent ? "true" : "false").Append(",");
            sb.Append("\"probeRequestJson\":").Append(Quote(probeRequestJson ?? string.Empty)).Append(",");
            sb.Append("\"outgoingAgentMessage\":").Append(Quote(Truncate(outgoingMessage ?? string.Empty, DiagnosticActivityTextMaxLength))).Append(",");
            sb.Append("\"initialActivityCount\":").Append(initialActivities == null ? 0 : initialActivities.Count).Append(",");
            sb.Append("\"responseActivityCount\":").Append(responseActivities == null ? 0 : responseActivities.Count).Append(",");
            sb.Append("\"validJsonResponseCount\":").Append(validJsonCount).Append(",");
            sb.Append("\"nonJsonBotMessageCount\":").Append(nonJsonBotMessageCount).Append(",");
            sb.Append("\"firstValidJson\":").Append(string.IsNullOrWhiteSpace(firstValidJson) ? "null" : Quote(firstValidJson)).Append(",");
            AppendDiagnosticActivitiesJson(sb, "initialActivities", initialActivities);
            sb.Append(",");
            AppendDiagnosticActivitiesJson(sb, "responseActivities", responseActivities);
            sb.Append(",");
            sb.Append("\"payload\":{\"header\":{},\"lines\":[]}");
            sb.Append("}");
            return sb.ToString();
        }

        private static void AppendDiagnosticActivitiesJson(StringBuilder sb, string propertyName, List<Activity> activities)
        {
            sb.Append(Quote(propertyName)).Append(":");
            sb.Append("[");

            if (activities != null)
            {
                for (int i = 0; i < activities.Count; i++)
                {
                    if (i > 0) sb.Append(",");

                    var a = activities[i];
                    if (a == null)
                    {
                        sb.Append("null");
                        continue;
                    }

                    string fromId = a.From == null ? string.Empty : (a.From.Id ?? string.Empty);
                    string fromName = a.From == null ? string.Empty : (a.From.Name ?? string.Empty);
                    string text = a.Text ?? string.Empty;
                    string extractedJson = TryExtractValidJson(text);
                    bool isFromPlugin = string.Equals(fromId, PluginSenderId, StringComparison.OrdinalIgnoreCase);

                    sb.Append("{");
                    sb.Append("\"id\":").Append(Quote(a.Id ?? string.Empty)).Append(",");
                    sb.Append("\"type\":").Append(Quote(a.Type ?? string.Empty)).Append(",");
                    sb.Append("\"fromId\":").Append(Quote(fromId)).Append(",");
                    sb.Append("\"fromName\":").Append(Quote(fromName)).Append(",");
                    sb.Append("\"isFromPlugin\":").Append(isFromPlugin ? "true" : "false").Append(",");
                    sb.Append("\"text\":").Append(Quote(Truncate(text, DiagnosticActivityTextMaxLength))).Append(",");
                    sb.Append("\"isValidJson\":").Append(string.IsNullOrWhiteSpace(extractedJson) ? "false" : "true").Append(",");
                    sb.Append("\"extractedJson\":").Append(string.IsNullOrWhiteSpace(extractedJson) ? "null" : Quote(extractedJson));
                    sb.Append("}");
                }
            }

            sb.Append("]");
        }

        private static string SendRequestToCopilotAgent(string baseUrl, string secret, string requestJson, string operation, int timeoutSeconds, ITracingService tracing)
        {
            var conversation = StartConversation(baseUrl, secret, timeoutSeconds, tracing);
            if (conversation == null || string.IsNullOrWhiteSpace(conversation.ConversationId))
            {
                return BuildApiErrorJson(operation, "Direct Line Error", "Failed to start Direct Line conversation.");
            }

            string watermark = null;

            // Drain any initial welcome or greeting activities so they do not get mistaken for the API response.
            try
            {
                Thread.Sleep(700);
                var initialActivities = GetActivities(baseUrl, secret, conversation.ConversationId, null, timeoutSeconds, tracing);
                if (initialActivities != null)
                {
                    watermark = initialActivities.Watermark;
                    tracing.Trace("Initial Direct Line activities drained. Watermark: " + (watermark ?? string.Empty));
                    TraceActivities(tracing, "Initial/drained activity", initialActivities.Activities);
                }
            }
            catch (Exception drainEx)
            {
                tracing.Trace("Initial activity drain warning: " + drainEx.Message);
            }

            string integrationMessage = BuildStrictApiModeMessage(requestJson);
            tracing.Trace("Exact message being sent to Copilot Agent. Length: " + integrationMessage.Length.ToString());
            TraceLarge(tracing, "Exact Copilot Agent outgoing message", integrationMessage);

            var activity = new OutboundActivity
            {
                Type = "message",
                From = new ChannelAccount
                {
                    Id = PluginSenderId,
                    Name = "Dynamics 365 Embedded Portal"
                },
                Text = integrationMessage
            };

            SendActivity(baseUrl, secret, conversation.ConversationId, activity, timeoutSeconds, tracing);

            DateTime stopAt = DateTime.UtcNow.AddSeconds(timeoutSeconds);
            string lastNonJsonBotText = string.Empty;
            int nonJsonMessageCount = 0;
            var nonJsonBotTexts = new List<string>();

            while (DateTime.UtcNow < stopAt)
            {
                Thread.Sleep(1000);

                var activitiesResponse = GetActivities(baseUrl, secret, conversation.ConversationId, watermark, timeoutSeconds, tracing);
                if (activitiesResponse == null)
                {
                    continue;
                }

                watermark = activitiesResponse.Watermark;

                if (activitiesResponse.Activities == null)
                {
                    continue;
                }

                TraceActivities(tracing, "Polled activity", activitiesResponse.Activities);

                foreach (var a in activitiesResponse.Activities)
                {
                    if (a == null) continue;
                    if (!string.Equals(a.Type, "message", StringComparison.OrdinalIgnoreCase)) continue;
                    if (string.IsNullOrWhiteSpace(a.Text)) continue;

                    bool isFromPlugin = a.From != null && string.Equals(a.From.Id, PluginSenderId, StringComparison.OrdinalIgnoreCase);
                    if (isFromPlugin) continue;

                    string text = a.Text.Trim();
                    tracing.Trace("Bot message received. Length: " + text.Length.ToString());

                    string json = TryExtractValidJson(text);
                    if (!string.IsNullOrWhiteSpace(json))
                    {
                        tracing.Trace("Valid JSON response found from agent.");
                        return json;
                    }

                    nonJsonMessageCount++;
                    lastNonJsonBotText = text;
                    if (nonJsonBotTexts.Count < 20)
                    {
                        nonJsonBotTexts.Add(text);
                    }
                    tracing.Trace("Non-JSON bot message ignored. Count: " + nonJsonMessageCount.ToString() + ". Text preview: " + Truncate(text, 250));

                    // Continue polling. Do not fail immediately because the first message may be a greeting.
                }
            }

            if (!string.IsNullOrWhiteSpace(lastNonJsonBotText))
            {
                return BuildInvalidAgentResponseJson(operation, lastNonJsonBotText, nonJsonBotTexts);
            }

            return BuildApiErrorJson(operation, "Timeout", "Timed out waiting for a valid JSON response from the Copilot Agent.");
        }

        private static string BuildStrictApiModeMessage(string requestJson)
        {
            // IMPORTANT:
            // The Copilot Agent expects the first line to be exactly:
            // CTP_VENDOR_QUOTE_AGENT_REQUEST
            // The JSON payload must start immediately after that marker.
            // Do not add labels such as REQUEST_JSON, system instructions, markdown, or extra prose here.
            string cleanJson = string.IsNullOrWhiteSpace(requestJson)
                ? "{}"
                : requestJson.Trim();

            return "CTP_VENDOR_QUOTE_AGENT_REQUEST\n" + cleanJson;
        }

        private static DirectLineConversation StartConversation(string baseUrl, string secret, int timeoutSeconds, ITracingService tracing)
        {
            string url = baseUrl + "/v3/directline/conversations";
            string response = HttpJson(url, "POST", secret, null, timeoutSeconds);
            tracing.Trace("Direct Line conversation started.");
            return Deserialize<DirectLineConversation>(response);
        }

        private static void SendActivity(string baseUrl, string secret, string conversationId, OutboundActivity activity, int timeoutSeconds, ITracingService tracing)
        {
            string url = baseUrl + "/v3/directline/conversations/" + Uri.EscapeDataString(conversationId) + "/activities";
            string body = Serialize(activity);
            tracing.Trace("Sending Direct Line activity. URL: " + url);
            TraceLarge(tracing, "Direct Line outgoing activity JSON body", body);
            HttpJson(url, "POST", secret, body, timeoutSeconds);
            tracing.Trace("Activity sent to Copilot Agent.");
        }

        private static ActivitiesResponse GetActivities(string baseUrl, string secret, string conversationId, string watermark, int timeoutSeconds, ITracingService tracing)
        {
            string url = baseUrl + "/v3/directline/conversations/" + Uri.EscapeDataString(conversationId) + "/activities";
            if (!string.IsNullOrWhiteSpace(watermark))
            {
                url += "?watermark=" + Uri.EscapeDataString(watermark);
            }

            tracing.Trace("Getting Direct Line activities. URL: " + url);
            string response = HttpJson(url, "GET", secret, null, timeoutSeconds);
            tracing.Trace("Direct Line activities response length: " + (response == null ? "0" : response.Length.ToString()));
            return Deserialize<ActivitiesResponse>(response);
        }

        private static string HttpJson(string url, string method, string bearerSecretOrToken, string jsonBody, int timeoutSeconds)
        {
            var request = (HttpWebRequest)WebRequest.Create(url);
            request.Method = method;
            request.ContentType = "application/json";
            request.Accept = "application/json";
            request.Headers["Authorization"] = "Bearer " + bearerSecretOrToken;
            request.KeepAlive = false;
            request.Timeout = timeoutSeconds * 1000;
            request.ReadWriteTimeout = timeoutSeconds * 1000;

            if (!string.IsNullOrWhiteSpace(jsonBody))
            {
                byte[] bytes = Encoding.UTF8.GetBytes(jsonBody);
                request.ContentLength = bytes.Length;
                using (var stream = request.GetRequestStream())
                {
                    stream.Write(bytes, 0, bytes.Length);
                }
            }

            try
            {
                using (var response = (HttpWebResponse)request.GetResponse())
                using (var responseStream = response.GetResponseStream())
                using (var reader = new StreamReader(responseStream))
                {
                    return reader.ReadToEnd();
                }
            }
            catch (WebException webEx)
            {
                string errorBody = string.Empty;
                if (webEx.Response != null)
                {
                    using (var stream = webEx.Response.GetResponseStream())
                    using (var reader = new StreamReader(stream))
                    {
                        errorBody = reader.ReadToEnd();
                    }
                }

                throw new InvalidPluginExecutionException("Direct Line HTTP call failed. " + webEx.Message + " " + errorBody, webEx);
            }
        }

        private static string GetInputString(IPluginExecutionContext context, string key)
        {
            if (!context.InputParameters.Contains(key) || context.InputParameters[key] == null)
            {
                return string.Empty;
            }

            return context.InputParameters[key].ToString();
        }

        private static string GetRequestJsonFromContext(IPluginExecutionContext context, ITracingService tracing)
        {
            // Primary expected Custom API request parameter.
            string value = GetInputString(context, InputRequestJson);
            if (!string.IsNullOrWhiteSpace(value))
            {
                tracing.Trace("Request was read from input parameter: " + InputRequestJson);
                return value;
            }

            // Defensive aliases for mistakes made during portal testing.
            string[] aliases = new[] { "RequestJson", "Request Json", "requestJson", "ctp_requestjson" };
            for (int i = 0; i < aliases.Length; i++)
            {
                value = GetInputString(context, aliases[i]);
                if (!string.IsNullOrWhiteSpace(value))
                {
                    tracing.Trace("Request was read from alias input parameter: " + aliases[i]);
                    return value;
                }
            }

            // Final defensive scan: if any string input contains an operation value, use it.
            foreach (KeyValuePair<string, object> item in context.InputParameters)
            {
                if (item.Value == null) continue;
                string text = item.Value.ToString();
                if (text.IndexOf("\"operation\"", StringComparison.OrdinalIgnoreCase) >= 0)
                {
                    tracing.Trace("Request was read by scanning input parameter: " + item.Key);
                    return text;
                }
            }

            return string.Empty;
        }

        private static void TraceInputParameterKeys(IPluginExecutionContext context, ITracingService tracing)
        {
            if (context == null || tracing == null) return;

            tracing.Trace("Input parameter count: " + context.InputParameters.Count.ToString());
            foreach (KeyValuePair<string, object> item in context.InputParameters)
            {
                string typeName = item.Value == null ? "null" : item.Value.GetType().FullName;
                int length = item.Value == null ? 0 : item.Value.ToString().Length;
                tracing.Trace("Input parameter: " + item.Key + ", type=" + typeName + ", stringLength=" + length.ToString());
            }
        }

        private static string NormalizeRequestJson(string raw, ITracingService tracing)
        {
            if (string.IsNullOrWhiteSpace(raw)) return string.Empty;

            string text = raw.Trim();
            if (text.Length > 0 && text[0] == '\uFEFF')
            {
                text = text.Substring(1).Trim();
            }

            // If someone accidentally passes the whole Web API body into ctp_RequestJson,
            // unwrap { "ctp_RequestJson": "{...}" }.
            string wrapped = TryExtractWrappedCtpRequestJson(text);
            if (!string.IsNullOrWhiteSpace(wrapped))
            {
                tracing.Trace("Detected and unwrapped nested ctp_RequestJson body.");
                text = wrapped.Trim();
            }

            // If the portal double-stringified the JSON, turn "{\"operation\":...}" back into {"operation":...}.
            for (int i = 0; i < 3; i++)
            {
                if (LooksLikeJsonObject(text)) break;

                string unescaped = TryDeserializeJsonStringLiteral(text);
                if (!string.IsNullOrWhiteSpace(unescaped) && !string.Equals(unescaped, text, StringComparison.Ordinal))
                {
                    tracing.Trace("Detected and normalized double-encoded JSON string. Pass: " + (i + 1).ToString());
                    text = unescaped.Trim();
                    continue;
                }

                break;
            }

            return text.Trim();
        }

        private static string TryExtractWrappedCtpRequestJson(string text)
        {
            if (string.IsNullOrWhiteSpace(text)) return null;
            if (!LooksLikeJsonObject(text)) return null;
            if (text.IndexOf("ctp_RequestJson", StringComparison.OrdinalIgnoreCase) < 0) return null;

            var match = Regex.Match(
                text,
                "\"ctp_RequestJson\"\\s*:\\s*\"(?<value>(?:\\\\.|[^\"])*)\"",
                RegexOptions.IgnoreCase | RegexOptions.Singleline);

            if (!match.Success) return null;

            string literal = "\"" + match.Groups["value"].Value + "\"";
            return TryDeserializeJsonStringLiteral(literal);
        }

        private static string TryDeserializeJsonStringLiteral(string text)
        {
            if (string.IsNullOrWhiteSpace(text)) return null;
            string t = text.Trim();
            if (!t.StartsWith("\"", StringComparison.Ordinal) || !t.EndsWith("\"", StringComparison.Ordinal)) return null;

            try
            {
                var serializer = new DataContractJsonSerializer(typeof(string));
                using (var ms = new MemoryStream(Encoding.UTF8.GetBytes(t)))
                {
                    return (string)serializer.ReadObject(ms);
                }
            }
            catch
            {
                return null;
            }
        }

        private static string GetEnvironmentVariable(IOrganizationService service, string schemaName, bool required = true)
        {
            var definitionQuery = new QueryExpression("environmentvariabledefinition")
            {
                ColumnSet = new ColumnSet("environmentvariabledefinitionid", "schemaname", "defaultvalue")
            };
            definitionQuery.Criteria.AddCondition("schemaname", ConditionOperator.Equal, schemaName);
            definitionQuery.TopCount = 1;

            var definitions = service.RetrieveMultiple(definitionQuery);
            if (definitions.Entities.Count == 0)
            {
                if (required)
                {
                    throw new InvalidPluginExecutionException("Missing environment variable definition: " + schemaName);
                }
                return null;
            }

            var definition = definitions.Entities[0];
            string defaultValue = definition.Contains("defaultvalue") ? definition.GetAttributeValue<string>("defaultvalue") : null;

            var valueQuery = new QueryExpression("environmentvariablevalue")
            {
                ColumnSet = new ColumnSet("value")
            };
            valueQuery.Criteria.AddCondition("environmentvariabledefinitionid", ConditionOperator.Equal, definition.Id);
            valueQuery.TopCount = 1;

            var values = service.RetrieveMultiple(valueQuery);
            if (values.Entities.Count > 0 && values.Entities[0].Contains("value"))
            {
                string currentValue = values.Entities[0].GetAttributeValue<string>("value");
                if (!string.IsNullOrWhiteSpace(currentValue))
                {
                    return currentValue;
                }
            }

            return defaultValue;
        }

        private static string TryExtractOperation(string requestJson)
        {
            if (string.IsNullOrWhiteSpace(requestJson)) return string.Empty;

            var match = Regex.Match(requestJson, "\"operation\"\\s*:\\s*\"(?<operation>[^\"]+)\"", RegexOptions.IgnoreCase);
            if (match.Success)
            {
                return match.Groups["operation"].Value;
            }

            return string.Empty;
        }

        private static string TryExtractValidJson(string text)
        {
            if (string.IsNullOrWhiteSpace(text)) return null;

            string cleaned = RemoveCodeFence(text.Trim());

            if (LooksLikeJsonObject(cleaned) && IsValidJson(cleaned))
            {
                return cleaned;
            }

            string candidate = ExtractFirstJsonObject(cleaned);
            if (!string.IsNullOrWhiteSpace(candidate) && IsValidJson(candidate))
            {
                return candidate;
            }

            return null;
        }

        private static string RemoveCodeFence(string text)
        {
            if (string.IsNullOrWhiteSpace(text)) return text;
            string t = text.Trim();

            if (t.StartsWith("```", StringComparison.Ordinal))
            {
                int firstNewLine = t.IndexOf('\n');
                if (firstNewLine >= 0)
                {
                    t = t.Substring(firstNewLine + 1);
                }

                int lastFence = t.LastIndexOf("```", StringComparison.Ordinal);
                if (lastFence >= 0)
                {
                    t = t.Substring(0, lastFence);
                }
            }

            return t.Trim();
        }

        private static bool LooksLikeJsonObject(string text)
        {
            if (string.IsNullOrWhiteSpace(text)) return false;
            string t = text.Trim();
            return t.StartsWith("{") && t.EndsWith("}");
        }

        private static bool IsValidJson(string json)
        {
            if (string.IsNullOrWhiteSpace(json)) return false;

            try
            {
                byte[] bytes = Encoding.UTF8.GetBytes(json);
                using (var ms = new MemoryStream(bytes))
                using (var reader = JsonReaderWriterFactory.CreateJsonReader(ms, XmlDictionaryReaderQuotas.Max))
                {
                    while (reader.Read()) { }
                    return true;
                }
            }
            catch
            {
                return false;
            }
        }

        private static string ExtractFirstJsonObject(string text)
        {
            if (string.IsNullOrWhiteSpace(text)) return null;

            int start = text.IndexOf('{');
            if (start < 0) return null;

            bool inString = false;
            bool escaped = false;
            int depth = 0;

            for (int i = start; i < text.Length; i++)
            {
                char c = text[i];

                if (escaped)
                {
                    escaped = false;
                    continue;
                }

                if (c == '\\' && inString)
                {
                    escaped = true;
                    continue;
                }

                if (c == '"')
                {
                    inString = !inString;
                    continue;
                }

                if (inString) continue;

                if (c == '{') depth++;
                if (c == '}') depth--;

                if (depth == 0)
                {
                    return text.Substring(start, i - start + 1).Trim();
                }
            }

            return null;
        }

        private static void TraceLarge(ITracingService tracing, string label, string value)
        {
            if (tracing == null) return;

            if (value == null)
            {
                tracing.Trace(label + ": [NULL]");
                return;
            }

            tracing.Trace(label + " length: " + value.Length.ToString());

            if (value.Length == 0)
            {
                tracing.Trace(label + ": [EMPTY]");
                return;
            }

            int chunk = 1;
            for (int i = 0; i < value.Length; i += TraceChunkSize)
            {
                int len = Math.Min(TraceChunkSize, value.Length - i);
                tracing.Trace(label + " [chunk " + chunk.ToString() + "]: " + value.Substring(i, len));
                chunk++;
            }
        }

        private static void TraceActivities(ITracingService tracing, string label, List<Activity> activities)
        {
            if (tracing == null) return;

            if (activities == null)
            {
                tracing.Trace(label + ": [NULL activities]");
                return;
            }

            tracing.Trace(label + " count: " + activities.Count.ToString());

            for (int i = 0; i < activities.Count; i++)
            {
                var a = activities[i];
                if (a == null)
                {
                    tracing.Trace(label + "[" + i.ToString() + "]: [NULL activity]");
                    continue;
                }

                string fromId = a.From == null ? string.Empty : (a.From.Id ?? string.Empty);
                string fromName = a.From == null ? string.Empty : (a.From.Name ?? string.Empty);
                string text = a.Text ?? string.Empty;
                tracing.Trace(label + "[" + i.ToString() + "] type=" + (a.Type ?? string.Empty)
                    + ", fromId=" + fromId
                    + ", fromName=" + fromName
                    + ", textLength=" + text.Length.ToString()
                    + ", textPreview=" + Truncate(text, 500));
            }
        }

        private static string BuildApiErrorJson(string operation, string status, string message)
        {
            return "{"
                + "\"success\":false,"
                + "\"operation\":" + Quote(operation ?? string.Empty) + ","
                + "\"status\":" + Quote(status ?? "Error") + ","
                + "\"message\":" + Quote(message ?? string.Empty) + ","
                + "\"payload\":{\"header\":{},\"lines\":[]}"
                + "}";
        }

        private static string BuildInvalidAgentResponseJson(string operation, string rawAgentResponse, List<string> allNonJsonResponses)
        {
            var sb = new StringBuilder();
            sb.Append("{");
            sb.Append("\"success\":false,");
            sb.Append("\"operation\":").Append(Quote(operation ?? string.Empty)).Append(",");
            sb.Append("\"status\":\"Invalid Agent Response\",");
            sb.Append("\"message\":\"The agent responded with text instead of valid JSON. See rawAgentResponse and rawAgentResponses for the captured response activities.\",");
            sb.Append("\"rawAgentResponse\":").Append(Quote(Truncate(rawAgentResponse, RawAgentResponseMaxLength))).Append(",");
            sb.Append("\"rawAgentResponses\":[");

            if (allNonJsonResponses != null)
            {
                for (int i = 0; i < allNonJsonResponses.Count; i++)
                {
                    if (i > 0) sb.Append(",");
                    sb.Append(Quote(Truncate(allNonJsonResponses[i], RawAgentResponseMaxLength)));
                }
            }

            sb.Append("],");
            sb.Append("\"payload\":{\"header\":{},\"lines\":[]}");
            sb.Append("}");
            return sb.ToString();
        }

        private static string Quote(string value)
        {
            if (value == null) return "null";
            return "\"" + value.Replace("\\", "\\\\").Replace("\"", "\\\"").Replace("\r", "\\r").Replace("\n", "\\n").Replace("\t", "\\t") + "\"";
        }

        private static string Truncate(string value, int maxLength)
        {
            if (string.IsNullOrEmpty(value)) return value;
            if (value.Length <= maxLength) return value;
            return value.Substring(0, maxLength) + "...";
        }

        private static string Serialize<T>(T value)
        {
            var serializer = new DataContractJsonSerializer(typeof(T));
            using (var ms = new MemoryStream())
            {
                serializer.WriteObject(ms, value);
                return Encoding.UTF8.GetString(ms.ToArray());
            }
        }

        private static T Deserialize<T>(string json)
        {
            var serializer = new DataContractJsonSerializer(typeof(T));
            using (var ms = new MemoryStream(Encoding.UTF8.GetBytes(json)))
            {
                return (T)serializer.ReadObject(ms);
            }
        }
